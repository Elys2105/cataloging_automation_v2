        private void FindAndOpenOneDocumentRecord(string vanBan)
        {
            AddLog("Tìm bản ghi riêng", "Đang làm", "Đang tìm bản ghi: " + vanBan);
            SetStatus("Đang tìm bản ghi " + vanBan + "...");

            string wanted = Normalize(vanBan);
            int pageIndex = 1;

            while (true)
            {
                WaitPage();

                var rows = driver.FindElements(By.XPath("//table//tbody//tr | //tr[td]"));

                foreach (var row in rows)
                {
                    try
                    {
                        string rowText = row.Text ?? "";
                        string normRow = Normalize(rowText);

                        if (!normRow.Contains(wanted))
                            continue;

                        IWebElement editLink = null;

                        var links = row.FindElements(By.XPath(
                            ".//a[contains(@href,'/DocumentCataloging/UpdateDoc') or contains(@href,'UpdateDoc?id=')]"
                        ));

                        if (links.Count > 0)
                        {
                            editLink = links[0];
                        }
                        else
                        {
                            var editButtons = row.FindElements(By.XPath(
                                ".//*[contains(@title,'Sửa') or contains(@data-original-title,'Sửa') or contains(@aria-label,'Sửa')]"
                            ));

                            if (editButtons.Count > 0)
                            {
                                editLink = editButtons[0].FindElement(By.XPath("./ancestor::a[contains(@href,'UpdateDoc')][1]"));
                            }
                        }

                        if (editLink == null)
                            continue;

                        string href = editLink.GetAttribute("href");

                        AddLog("Tìm bản ghi riêng", "OK", "Đã tìm thấy bản ghi ở trang " + pageIndex + ": " + rowText);

                        if (!string.IsNullOrWhiteSpace(href))
                        {
                            driver.Navigate().GoToUrl(href);
                        }
                        else
                        {
                            ClickElement(editLink);
                        }

                        WaitPage();

                        wait.Until(d =>
                            d.Url.Contains("/UpdateDoc") ||
                            d.FindElement(By.TagName("body")).Text.Contains("Thông tin tài liệu")
                        );

                        AddLog("Mở bản ghi riêng", "OK", "Đã mở bản ghi cần sửa.");
                        return;
                    }
                    catch
                    {
                    }
                }

                AddLog("Tìm bản ghi riêng", "Đang làm", "Chưa thấy ở trang " + pageIndex + ", kiểm tra trang tiếp theo.");

                if (!GoNextPage())
                    break;

                pageIndex++;

                if (pageIndex > 100)
                    throw new Exception("Dừng tìm bản ghi riêng vì vượt quá 100 trang.");
            }

            throw new Exception("Không tìm thấy bản ghi cần chạy riêng: " + vanBan);
        }

        private void StartChromeNew()
        {
            ChromeOptions options = new ChromeOptions();

            options.AddArgument("--start-maximized");
            options.AddArgument("--disable-notifications");
            options.AddArgument("--disable-popup-blocking");

            /*
             * Dùng profile riêng để lần sau khỏi đăng nhập lại.
             * Lần chạy đầu tiên, Chrome mở lên thì bạn đăng nhập tài khoản.
             * Sau đó chạy lại automation, nó sẽ nhớ đăng nhập.
             */
            string profilePath = Path.Combine(
                Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
                "KhoLuuTruAutomationChromeProfile"
            );

            options.AddArgument("--user-data-dir=" + profilePath);
            options.AddArgument("--disable-extensions");
            options.AddArgument("--disable-background-networking");
            options.AddArgument("--disable-sync");
            options.AddArgument("--disable-translate");
            options.AddArgument("--disable-features=Translate,BackForwardCache");
            driver = new ChromeDriver(options);
            wait = new WebDriverWait(driver, TimeSpan.FromSeconds(25));
        }

        private void SearchAndOpenProfileByTitle(string hoSo)
        {
            AddLog("Tìm hồ sơ", "Đang làm", hoSo);
            SetStatus("Đang tìm hồ sơ " + hoSo + "...");

            IWebElement searchBox = WaitVisible(By.XPath(
                "//input[contains(@placeholder,'Nhập') or contains(@placeholder,'Số') or contains(@placeholder,'hồ sơ') or contains(@placeholder,'Tiêu đề')]"
            ));

            ClearAndType(searchBox, hoSo);

            IWebElement btnSearch = WaitClickable(By.XPath(
                "//button[contains(normalize-space(),'Tìm kiếm')] | //a[contains(normalize-space(),'Tìm kiếm')]"
            ));

            ClickElement(btnSearch);
            WaitPage();

            AddLog("Tìm hồ sơ", "OK", "Đã bấm tìm kiếm.");

            IWebElement row = WaitVisible(By.XPath(
                "//tr[.//td[contains(normalize-space(),'" + hoSo + "')]]"
            ));

            AddLog("Mở hồ sơ", "Đang làm", "Đã tìm thấy hồ sơ " + hoSo);

            IWebElement titleLink = null;

            // Ưu tiên click đúng cột TIÊU ĐỀ HỒ SƠ, thường là cột thứ 3 hoặc có chữ 'Tập lưu'
            var titleLinks = row.FindElements(By.XPath(
                ".//td[3]//a | " +
                ".//a[contains(normalize-space(),'Tập lưu')] | " +
                ".//a[contains(@href,'ListDocument')]"
            ));

            if (titleLinks.Count > 0)
            {
                titleLink = titleLinks[0];
            }
            else
            {
                // Trường hợp tiêu đề không phải thẻ a nhưng là chữ xanh/có onclick
                var titleTexts = row.FindElements(By.XPath(
                    ".//td[3]//*[contains(normalize-space(),'Tập lưu')] | " +
                    ".//*[contains(@class,'text-primary') or contains(@class,'blue') or contains(@style,'blue')]"
                ));

                if (titleTexts.Count > 0)
                    titleLink = titleTexts[0];
            }

            if (titleLink == null)
                throw new Exception("Không tìm thấy tiêu đề hồ sơ màu xanh để mở hồ sơ: " + hoSo);

            ClickElement(titleLink);
            WaitPage();

            wait.Until(d =>
                d.Url.Contains("/ListDocument") ||
                d.FindElement(By.TagName("body")).Text.Contains("Danh sách thành phần hồ sơ")
            );

            AddLog("Mở hồ sơ", "OK", "Đã mở trang danh sách thành phần hồ sơ.");
        }

        private void ProcessAllDocumentRecords()
        {
            int processed = 0;
            int pageIndex = 1;
            var processedUrls = new System.Collections.Generic.HashSet<string>();

            while (true)
            {
                WaitPage();

                string currentListUrl = driver.Url;

                AddLog("Trang danh sách", "Đang làm", "Đang đọc danh sách bản ghi trang " + pageIndex);

                var editLinks = GetEditLinksOnCurrentPage();

                if (editLinks.Count == 0)
                {
                    AddLog("Trang danh sách", "Cảnh báo", "Không tìm thấy nút Sửa nào trên trang hiện tại.");
                    break;
                }

                AddLog("Trang danh sách", "OK", "Tìm thấy " + editLinks.Count + " bản ghi cần xử lý trên trang " + pageIndex);

                for (int i = 0; i < editLinks.Count; i++)
                {
                    string editUrl = editLinks[i];

                    if (processedUrls.Contains(editUrl))
                    {
                        AddLog("Bỏ qua bản ghi", "OK", "Bản ghi này đã xử lý rồi, không chạy lại: " + editUrl);
                        continue;
                    }

                    processedUrls.Add(editUrl);
                    processed++;

                    AddLog("Bản ghi " + processed, "Đang làm", editUrl);
                    SetStatus("Đang xử lý bản ghi " + processed + "...");

                    driver.Navigate().GoToUrl(editUrl);
                    WaitPage();

                    ProcessOneDocumentRecord();
                    GoBackToDocumentList();
                }

                // Quan trọng: sau khi xử lý xong các bản ghi của trang hiện tại,
                // quay lại đúng URL trang danh sách đang xử lý rồi mới kiểm tra trang tiếp theo.
                // Tránh trường hợp web tự trả về pageIndex=1 làm chương trình chạy lại bản ghi cũ.
                driver.Navigate().GoToUrl(currentListUrl);
                WaitPage();

                if (!GoNextPage())
                {
                    break;
                }

                pageIndex++;
            }

            AddLog("Tổng kết", "OK", "Đã xử lý " + processed + " bản ghi.");
        }

        {
            public string Url { get; set; }
            public string RowText { get; set; }
        }

        private void ProcessDocumentRecordsFromVanBan(string vanBan)
        {
            string wanted = Normalize(vanBan);
            bool started = false;
            int processed = 0;
            int pageIndex = 1;
            var processedUrls = new System.Collections.Generic.HashSet<string>();

            while (true)
            {
                CheckStopRequested();
                WaitPage();

                string currentListUrl = driver.Url;

                AddLog("Tiếp tục danh sách", "Đang làm", "Đang kiểm tra trang " + pageIndex + " để tìm bản ghi bắt đầu: " + vanBan);

                var items = GetDocumentListItemsOnCurrentPage();

                if (items.Count == 0)
                {
                    AddLog("Tiếp tục danh sách", "Cảnh báo", "Không tìm thấy bản ghi/nút Sửa nào trên trang hiện tại.");
                }

                foreach (var item in items)
                {
                    CheckStopRequested();

                    if (!started)
                    {
                        string rowKey = Normalize(item.RowText);
                        string urlKey = Normalize(item.Url);

                        if (!rowKey.Contains(wanted) && !urlKey.Contains(wanted))
                            continue;

                        started = true;
                        AddLog("Tiếp tục từ bản ghi", "OK", "Bắt đầu chạy từ dòng: " + item.RowText);
                    }

                    if (processedUrls.Contains(item.Url))
                    {
                        AddLog("Bỏ qua bản ghi", "OK", "Bản ghi này đã xử lý trong lần chạy tiếp tục: " + item.Url);
                        continue;
                    }

                    processedUrls.Add(item.Url);
                    processed++;

                    AddLog("Bản ghi tiếp tục " + processed, "Đang làm", item.Url);
                    SetStatus("Đang chạy tiếp bản ghi " + processed + "...");

                    driver.Navigate().GoToUrl(item.Url);
                    WaitPage();

                    ProcessOneDocumentRecord();

                    GoBackToDocumentList();

                    driver.Navigate().GoToUrl(currentListUrl);
                    WaitPage();
                }

                if (!GoNextPage())
                    break;

                pageIndex++;

                if (pageIndex > 100)
                    throw new Exception("Dừng tiếp tục vì số trang vượt quá giới hạn an toàn.");
            }

            if (!started)
                throw new Exception("Không tìm thấy bản ghi bắt đầu để tiếp tục: " + vanBan);

            if (processed == 0)
                throw new Exception("Đã tìm thấy bản ghi bắt đầu nhưng chưa xử lý được bản ghi nào.");

            AddLog("Tổng kết tiếp tục", "OK", "Đã xử lý " + processed + " bản ghi từ [" + vanBan + "] tới hết danh sách.");
        }

        private System.Collections.Generic.List<DocumentListItem> GetDocumentListItemsOnCurrentPage()
        {
            var result = new System.Collections.Generic.List<DocumentListItem>();

            WaitPage();

            try
            {
                ((IJavaScriptExecutor)driver).ExecuteScript(
                    "var el = document.querySelector('.dataTables_scrollBody, .table-responsive, div[style*=\"overflow\"]');" +
                    "if (el) el.scrollLeft = el.scrollWidth;"
                );
                Thread.Sleep(300);
            }
            catch
            {
            }

            var rows = driver.FindElements(By.XPath("//table//tbody//tr | //tr[td]"));

            foreach (var row in rows)
            {
                try
                {
                    string rowText = row.Text ?? "";
                    string href = "";

                    var links = row.FindElements(By.XPath(
                        ".//a[contains(@href,'/DocumentCataloging/UpdateDoc') or contains(@href,'UpdateDoc?id=')]"
                    ));

                    if (links.Count > 0)
                    {
                        href = links[0].GetAttribute("href");
                    }

                    if (string.IsNullOrWhiteSpace(href))
                    {
                        var editButtons = row.FindElements(By.XPath(
                            ".//*[contains(@title,'Sửa') or contains(@data-original-title,'Sửa') or contains(@aria-label,'Sửa')]"
                        ));

                        foreach (var btn in editButtons)
                        {
                            try
                            {
                                IWebElement a = btn.FindElement(By.XPath("./ancestor::a[contains(@href,'UpdateDoc')][1]"));
                                href = a.GetAttribute("href");
                                if (!string.IsNullOrWhiteSpace(href))
                                    break;
                            }
                            catch
                            {
                            }
                        }
                    }

                    if (string.IsNullOrWhiteSpace(href))
                        continue;

                    if (result.Any(x => x.Url == href))
                        continue;

                    result.Add(new DocumentListItem
                    {
                        Url = href,
                        RowText = rowText
                    });
                }
                catch
                {
                }
            }

            return result;
        }
        private class PageCompletionInfo
        {
            public int Total { get; set; }
            public int Completed { get; set; }
            public string BadRows { get; set; }
        }

        private void VerifyAllDocumentStatusCompleted()
        {
            AddLog("Kiểm tra trạng thái", "Đang làm", "Kiểm tra lại toàn bộ bản ghi sau khi chạy xong.");
            SetStatus("Đang kiểm tra trạng thái Hoàn thành...");

            GoFirstPageOfDocumentList();

            int pageIndex = 1;
            int totalRows = 0;
            int totalCompleted = 0;

            while (true)
            {
                WaitPage();

                PageCompletionInfo info = GetCompletionInfoOnCurrentPage();

                if (info.Total == 0)
                    throw new Exception("Không đọc được danh sách bản ghi để kiểm tra trạng thái Hoàn thành.");

                totalRows += info.Total;
                totalCompleted += info.Completed;

                AddLog(
                    "Kiểm tra trang " + pageIndex,
                    info.Total == info.Completed ? "OK" : "FAILED",
                    "Hoàn thành " + info.Completed + "/" + info.Total +
                    (string.IsNullOrWhiteSpace(info.BadRows) ? "" : ". Dòng chưa hoàn thành: " + info.BadRows)
                );

                if (info.Total != info.Completed)
                {
                    throw new Exception(
                        "Chưa xong: trang " + pageIndex +
                        " còn bản ghi chưa có trạng thái Hoàn thành. Dòng: " + info.BadRows
                    );
                }

                if (!GoNextPage())
                    break;

                pageIndex++;

                if (pageIndex > 100)
                    throw new Exception("Dừng kiểm tra vì số trang vượt quá giới hạn an toàn.");
            }

            SetStatus("Đã xong. Tất cả " + totalCompleted + "/" + totalRows + " bản ghi đều Hoàn thành.");
            AddLog("Kiểm tra trạng thái", "OK", "Đã xong. Tất cả " + totalCompleted + "/" + totalRows + " bản ghi đều Hoàn thành.");
        }

        private void GoFirstPageOfDocumentList()
        {
            WaitPage();

            try
            {
                string url = driver.Url;

                if (url.Contains("/DocumentCataloging/ListDocument") && Regex.IsMatch(url, @"pageIndex=\d+"))
                {
                    string firstPageUrl = Regex.Replace(url, @"pageIndex=\d+", "pageIndex=1");

                    if (firstPageUrl != url)
                    {
                        driver.Navigate().GoToUrl(firstPageUrl);
                        WaitPage();
                        AddLog("Kiểm tra trạng thái", "OK", "Đã quay về trang đầu danh sách để kiểm tra.");
                        return;
                    }
                }
            }
            catch
            {
            }

            try
            {
                var firstButtons = driver.FindElements(By.XPath(
                    "//a[contains(normalize-space(),'Trang đầu') or normalize-space()='«'] | " +
                    "//button[contains(normalize-space(),'Trang đầu') or normalize-space()='«']"
                ));

                foreach (var btn in firstButtons)
                {
                    try
                    {
                        string cls = btn.GetAttribute("class") ?? "";
                        string disabled = btn.GetAttribute("disabled") ?? "";

                        if (cls.ToLower().Contains("disabled") || disabled.ToLower() == "true")
                            continue;

                        if (btn.Displayed && btn.Enabled)
                        {
                            ClickElement(btn);
                            WaitPage();
                            AddLog("Kiểm tra trạng thái", "OK", "Đã bấm Trang đầu để kiểm tra.");
                            return;
                        }
                    }
                    catch
                    {
                    }
                }
            }
            catch
            {
            }
        }

        private PageCompletionInfo GetCompletionInfoOnCurrentPage()
        {
            PageCompletionInfo info = new PageCompletionInfo();

            try
            {
                object value = ((IJavaScriptExecutor)driver).ExecuteScript(@"
                    function visible(el) {
                        if (!el) return false;
                        const r = el.getBoundingClientRect();
                        return r.width > 0 && r.height > 0;
                    }

                    function norm(s) {
                        return (s || '')
                            .toLowerCase()
                            .replace(/\s+/g, ' ')
                            .trim();
                    }

                    let rows = Array.from(document.querySelectorAll('table tbody tr'))
                        .filter(r => visible(r) && r.querySelectorAll('td').length >= 5);

                    if (rows.length === 0) {
                        rows = Array.from(document.querySelectorAll('tr'))
                            .filter(r => visible(r) && r.querySelectorAll('td').length >= 5);
                    }

                    let total = rows.length;
                    let completed = 0;
                    let badRows = [];

                    rows.forEach((r, index) => {
                        const text = norm(r.innerText || r.textContent);

                        if (text.includes('hoàn thành')) {
                            completed++;
                        } else {
                            badRows.push(index + 1);
                        }
                    });

                    return total + '|' + completed + '|' + badRows.join(',');
                ");

                string raw = value == null ? "0|0|" : value.ToString();
                string[] parts = raw.Split('|');

                if (parts.Length >= 2)
                {
                    int total;
                    int completed;

                    int.TryParse(parts[0], out total);
                    int.TryParse(parts[1], out completed);

                    info.Total = total;
                    info.Completed = completed;
                    info.BadRows = parts.Length >= 3 ? parts[2] : "";
                }
            }
            catch (Exception ex)
            {
                AddLog("Kiểm tra trạng thái", "FAILED", ex.Message);
            }

            return info;
        }

        private void ClearTrichYeuNoiDung()
        {
            try
            {
                ((IJavaScriptExecutor)driver).ExecuteScript(@"
                    const wanted = 'trích yếu nội dung';

                    function norm(s) {
                        return (s || '')
                            .toLowerCase()
                            .replace(/\*/g, '')
                            .replace(/\s+/g, ' ')
                            .trim();
                    }

                    function hasBox(el) {
                        if (!el) return false;
                        const r = el.getBoundingClientRect();
                        return r.width > 0 && r.height > 0;
                    }

                    function isEditable(el) {
                        if (!el) return false;
                        const tag = (el.tagName || '').toLowerCase();
                        if (tag !== 'input' && tag !== 'textarea') return false;
                        const type = (el.getAttribute('type') || '').toLowerCase();
                        if (type === 'hidden' || type === 'button' || type === 'submit') return false;
                        if (el.disabled || el.readOnly) return false;
                        return hasBox(el);
                    }

                    const labelEls = Array.from(document.querySelectorAll('label, span, td, th, div'))
                        .filter(el => {
                            if (!hasBox(el)) return false;
                            const t = norm(el.innerText || el.textContent);
                            return t.includes(wanted) && t.length <= 80;
                        })
                        .sort((a, b) => {
                            const ra = a.getBoundingClientRect();
                            const rb = b.getBoundingClientRect();
                            return (ra.width * ra.height) - (rb.width * rb.height);
                        });

                    const fields = Array.from(document.querySelectorAll('textarea, input')).filter(isEditable);

                    for (const labelEl of labelEls) {
                        const lr = labelEl.getBoundingClientRect();
                        const labelCenterY = lr.top + lr.height / 2;

                        let candidates = fields
                            .map(f => ({ f: f, r: f.getBoundingClientRect(), tag: (f.tagName || '').toLowerCase() }))
                            .filter(x => {
                                const fieldCenterY = x.r.top + x.r.height / 2;
                                return x.r.left > lr.left &&
                                       x.r.top >= lr.top - 45 &&
                                       fieldCenterY <= labelCenterY + 120;
                            })
                            .sort((a, b) => {
                                if (a.tag === 'textarea' && b.tag !== 'textarea') return -1;
                                if (a.tag !== 'textarea' && b.tag === 'textarea') return 1;
                                const da = Math.abs(a.r.top - lr.top) + Math.abs(a.r.left - lr.right);
                                const db = Math.abs(b.r.top - lr.top) + Math.abs(b.r.left - lr.right);
                                return da - db;
                            });

                        if (candidates.length === 0) continue;

                        const el = candidates[0].f;
                        const tag = (el.tagName || '').toLowerCase();
                        const proto = tag === 'textarea'
                            ? HTMLTextAreaElement.prototype
                            : HTMLInputElement.prototype;
                        const setter = Object.getOwnPropertyDescriptor(proto, 'value').set;

                        setter.call(el, '');
                        el.removeAttribute('value');
                        el.dispatchEvent(new Event('input', { bubbles: true }));
                        el.dispatchEvent(new Event('change', { bubbles: true }));
                        el.dispatchEvent(new Event('blur', { bubbles: true }));
                        break;
                    }
                ");
            }
            catch
            {
                // Không throw — clear thất bại thì để SetTrichYeuNoiDung tự xử lý.
            }

            // Fallback Selenium: click vào textarea rồi Ctrl+A + Delete để đảm bảo xóa sạch.
            try
            {
                var fields = driver.FindElements(By.XPath(
                    "//*[contains(normalize-space(),'Trích yếu nội dung')]/following::textarea[not(@disabled) and not(@readonly)][1] | " +
                    "//*[contains(normalize-space(),'Trích yếu nội dung')]/following::input[not(@type='hidden') and not(@disabled) and not(@readonly)][1]"
                ));

                if (fields.Count > 0 && fields[0].Displayed && fields[0].Enabled)
                {
                    fields[0].Click();
                    Thread.Sleep(100);
                    fields[0].SendKeys(SeleniumKeys.Control + "a");
                    fields[0].SendKeys(SeleniumKeys.Delete);
                }
            }
            catch
            {
            }

            Thread.Sleep(200);
        }

        private void ProcessOneDocumentRecord()
        {
            AddLog("Xử lý tài liệu", "Đang làm", driver.Url);
            SetStatus("Đang kiểm tra số văn bản, ký hiệu văn bản, tác giả, độ mật...");

            // 1. Đọc số văn bản + ký hiệu văn bản từ file bên phải
            DocNumberInfo info = ExtractDocNumberInfoFromViewer();

            // FIX_OCR_EMPTY_USE_EXISTING_FORM_VALUES_CALL
            // Bổ sung nhẹ: nếu OCR local bị rỗng/yếu nhưng form hiện tại đã có dữ liệu,
            // dùng lại dữ liệu đang có trên form để không dừng nhầm.
            // Không đổi pipeline OCR/Selenium hiện tại.
            info = FillMissingInfoFromCurrentFormWhenOcrWeak(info);

            if (info == null ||
    (
        string.IsNullOrWhiteSpace(info.SoVanBan) &&
        string.IsNullOrWhiteSpace(info.KyHieuVanBan) &&
        string.IsNullOrWhiteSpace(info.TrichYeuNoiDung)
    ))
            {
                throw new Exception("OCR local chưa lấy được dữ liệu từ PDF bên phải. Dừng để tránh bấm Hoàn thành sai.");
            }

            if (!string.IsNullOrWhiteSpace(info.SoVanBan))
            {
                SetInputByLabelRobust("Số văn bản", info.SoVanBan);
                AddLog("Số văn bản", "OK", "Đã sửa theo OCR toàn văn: " + info.SoVanBan);
            }
            else
            {
                AddLog("Số văn bản", "Cảnh báo", "Không đọc được số văn bản từ OCR toàn văn.");
            }

            if (!string.IsNullOrWhiteSpace(info.KyHieuVanBan))
            {
                SetInputByLabelRobust("Ký hiệu văn bản", info.KyHieuVanBan);
                AddLog("Ký hiệu văn bản", "OK", "Đã sửa theo OCR toàn văn: " + info.KyHieuVanBan);
            }
            else
            {
                AddLog("Ký hiệu văn bản", "Cảnh báo", "Không đọc được ký hiệu văn bản từ OCR toàn văn.");
            }
            // BỔ SUNG: Tự chọn Tên thể loại văn bản theo ký hiệu hoặc nội dung OCR.
            // Ví dụ: 237-CV/ĐU => Công văn, QĐ => Quyết định, BC => Báo cáo...
            EnsureTenTheLoaiVanBan(info);
            // BẮT BUỘC xóa trích yếu cũ trước — kể cả khi extraction thất bại,
            // không để giá trị sai từ bản ghi trước còn sót lại.
            ClearTrichYeuNoiDung();

            if (!string.IsNullOrWhiteSpace(info.TrichYeuNoiDung))
            {
                SetTrichYeuNoiDung(info.TrichYeuNoiDung);
                AddLog("Trích yếu nội dung", "OK", "Đã sửa theo file: " + info.TrichYeuNoiDung);
            }
            else
            {
                AddLog("Trích yếu nội dung", "Cảnh báo", "Không đọc được trích yếu nội dung từ file bên phải.");
            }
            // 2. BẮT BUỘC sửa tác giả văn bản cố định
            SetTacGiaVanBanBatBuoc();
            AddLog("Tác giả văn bản", "OK", TAC_GIA_VAN_BAN_BAT_BUOC);

            // 3. Độ mật luôn chọn Thường
            SelectByLabelRobust("Độ mật", "Thường");
            AddLog("Độ mật", "OK", "Thường");

            // 4. Bỏ hoàn toàn phần Bản chính / Bản sao theo yêu cầu mới
            // Không xử lý Loại bản nữa.

            // 5. BẮT BUỘC tự động bấm Hoàn thành
            CompleteCurrentDocumentAndWaitSuccess();
        }

        // FIX_OCR_EMPTY_USE_EXISTING_FORM_VALUES_METHOD
        // Chỉ bổ sung: OCR yếu/rỗng thì dùng dữ liệu đang có trên form nếu hợp lệ.
        // Không sửa OCR, không sửa pipeline, không bấm nếu form cũng không có dữ liệu.
        private DocNumberInfo FillMissingInfoFromCurrentFormWhenOcrWeak(DocNumberInfo info)
        {
            if (info == null)
                info = new DocNumberInfo();

            bool changed = false;

            try
            {
                if (string.IsNullOrWhiteSpace(info.SoVanBan))
                {
                    string currentSo = GetCurrentValueByLabelRobust("Số văn bản");

                    if (IsUsableExistingShortValue(currentSo))
                    {
                        info.SoVanBan = currentSo.Trim();
                        changed = true;
                    }
                }

                if (string.IsNullOrWhiteSpace(info.KyHieuVanBan))
                {
                    string currentKy = GetCurrentValueByLabelRobust("Ký hiệu văn bản");

                    if (IsUsableExistingShortValue(currentKy))
                    {
                        info.KyHieuVanBan = NormalizeKyHieu(currentKy.Trim());
                        changed = true;
                    }
                }

                if (string.IsNullOrWhiteSpace(info.TrichYeuNoiDung))
                {
                    string currentTrichYeu = GetCurrentValueByLabelRobust("Trích yếu nội dung");

                    if (IsUsableExistingTrichYeu(currentTrichYeu))
                    {
                        info.TrichYeuNoiDung = CleanFinalTrichYeuTitle(currentTrichYeu.Trim());
                        changed = true;
                    }
                }

                if (string.IsNullOrWhiteSpace(info.TenTheLoaiVanBan))
                {
                    string currentType = GetCurrentValueByLabelRobust("Tên thể loại văn bản");

                    if (IsUsableExistingShortValue(currentType))
                    {
                        info.TenTheLoaiVanBan = currentType.Trim();
                        changed = true;
                    }
                }

                if (changed)
                {
                    AddLog(
                        "OCR fallback",
                        "OK",
                        "OCR local yếu/rỗng, đã dùng dữ liệu đang có trên form để không dừng nhầm."
                    );
                }
            }
            catch (Exception ex)
            {
                AddLog("OCR fallback", "Cảnh báo", "Không đọc được dữ liệu đang có trên form: " + ex.Message);
            }

            return info;
        }

        private bool IsUsableExistingShortValue(string value)
        {
            if (string.IsNullOrWhiteSpace(value))
                return false;

            string s = value.Trim();

            if (s == "..." || s == "-" || s == "--")
                return false;

            if (IsUiNoiseLine(s))
                return false;

            if (s.Length > 80)
                return false;

            return true;
        }

        private bool IsUsableExistingTrichYeu(string value)
        {
            if (string.IsNullOrWhiteSpace(value))
                return false;

            string s = CleanFinalTrichYeuTitle(value.Trim());

            if (s.Length < 6)
                return false;

            if (IsUiNoiseLine(s))
                return false;

            string key = FoldVietnameseKey(s);

            if (key.Contains("TRICH YEU NOI DUNG"))
                return false;

            if (key.StartsWith("CAN CU ") ||
                key.StartsWith("KINH GUI ") ||
                key.StartsWith("NOI NHAN ") ||
                key.StartsWith("T/M ") ||
                key.StartsWith("TM "))
                return false;

            return true;
        }

            SetStatus("Đang bấm Hoàn thành...");

            bool clicked = false;

            try
            {
                object jsClicked = ((IJavaScriptExecutor)driver).ExecuteScript(@"
            function visible(el) {

        private void ApplyStrictCongVanTrichYeuRule(DocNumberInfo result, string rawOcrText)
        {
            if (result == null)
                return;

            bool isCongVan =
                DocTypeEquals(result.TenTheLoaiVanBan, "Công văn") ||
                Regex.IsMatch(result.KyHieuVanBan ?? "",
                    @"(^|[-\s])CV\s*/\s*(ĐU|DU|VPĐU|VPDU|VP\s*ĐU|VP\s*DU)\b",
                    RegexOptions.IgnoreCase) ||
                Regex.IsMatch(rawOcrText ?? "",
                    @"S[oố0]\s*[:：]?\s*\d{1,6}\s*[-–—]\s*CV\s*/\s*(ĐU|DU|VPĐU|VPDU|VP\s*ĐU|VP\s*DU)\b",
                    RegexOptions.IgnoreCase);

            if (!isCongVan)
                return;

            string strictTitle = ExtractStrictCongVanTitleBetweenSoAndKinhGui(rawOcrText);

            if (string.IsNullOrWhiteSpace(strictTitle))
                strictTitle = ExtractStrictCongVanTitleBetweenSoAndKinhGui(result.TrichYeuNoiDung);

            if (!string.IsNullOrWhiteSpace(strictTitle))
            {
                result.TrichYeuNoiDung = strictTitle;

                AddLog(
                    "Công văn trích yếu",
                    "OK",
                    "Đã lấy riêng phần sau Số ...-CV/... và trước Kính gửi: " + result.TrichYeuNoiDung
                );

                return;
            }

            // Fallback an toàn: nếu OCR chỉ còn dạng "Số 403-CV/ĐU Về việc..."
            // thì vẫn cắt bỏ phần Số/Ký hiệu bằng hàm sẵn có.
            result.TrichYeuNoiDung = CleanCongVanTrichYeuFinal(result.TrichYeuNoiDung);
        }

        private string ExtractStrictCongVanTitleBetweenSoAndKinhGui(string rawText)
        {
            if (string.IsNullOrWhiteSpace(rawText))
                return "";

            string text = rawText
                .Replace("\r\n", "\n")
                .Replace("\r", "\n")
                .Replace("\u00A0", " ");

            text = Regex.Replace(text, @"[ \t]+", " ");

            string headerPattern =
                @"S[oố0]\s*[:：]?\s*\d{1,6}\s*[-–—]\s*CV\s*/\s*(?:ĐU|DU|VPĐU|VPDU|VP\s*ĐU|VP\s*DU)\b";

            // Ưu tiên đúng yêu cầu: lấy từ sau "Số ...-CV/..." đến trước "Kính gửi".
            Match m = Regex.Match(
                text,
                @"(?im)^\s*" + headerPattern + @"(?<after>[^\n]*)\n?(?<tail>[\s\S]*?)(?=^\s*(?:Kính\s*gửi|Kinh\s*gui|Kính\s*gui|Kinh\s*gửi)\b)",
                RegexOptions.IgnoreCase | RegexOptions.Multiline
            );

            if (m.Success)
            {
                string title = (m.Groups["after"].Value + "\n" + m.Groups["tail"].Value).Trim();
                title = CleanStrictCongVanTitleCandidate(title);

                if (!string.IsNullOrWhiteSpace(title))
                    return title;
            }

            // Fallback nhẹ: nếu OCR làm "Số 403-CV/ĐU Về việc..." nằm chung 1 dòng
            // nhưng không đọc được dòng Kính gửi trong crop.
            Match sameLine = Regex.Match(
                text,
                headerPattern + @"\s+(?<title>(?:Về|V/v|V\/v|Về việc|Báo cáo về)\b[^\n]*)",
                RegexOptions.IgnoreCase
            );

            if (sameLine.Success)
            {
                string title = CleanStrictCongVanTitleCandidate(sameLine.Groups["title"].Value);

                if (!string.IsNullOrWhiteSpace(title))
                    return title;
            }

            // Fallback cuối: nếu chuỗi truyền vào đã là trích yếu nhưng bị dính Số ...-CV/...
            string cleaned = CleanStrictCongVanTitleCandidate(text);

            if (!string.IsNullOrWhiteSpace(cleaned) &&
                Regex.IsMatch(cleaned, @"^(Về|V/v|V\/v|Về việc|Báo cáo về)\b", RegexOptions.IgnoreCase))
            {
                return cleaned;
            }

            return "";
        }

        private string CleanStrictCongVanTitleCandidate(string value)
        {
            if (string.IsNullOrWhiteSpace(value))
                return "";

            string s = value.Trim();

            s = s.Replace("&quot;", "\"")
                 .Replace("&amp;quot;", "\"")
                 .Replace("\u201C", "\"")
                 .Replace("\u201D", "\"")
                 .Replace("\u2018", "'")
                 .Replace("\u2019", "'");

            s = s.Replace("\r", "\n");

            // Nếu phần Kính gửi bị dính cùng dòng thì cắt tại đây.
            s = Regex.Replace(
                s,
                @"\s+(?=Kính\s*gửi|Kinh\s*gui|Kính\s*gui|Kinh\s*gửi).*$",
                "",
                RegexOptions.IgnoreCase | RegexOptions.Singleline
            );

            // Cắt phần Số ...-CV/ĐU hoặc Số ...-CV/VPĐU nếu OCR dính vào đầu trích yếu.
            s = Regex.Replace(
                s,
                @"^\s*S[oố0]\s*[:：]?\s*\d{1,6}\s*[-–—]\s*CV\s*/\s*(?:ĐU|DU|VPĐU|VPDU|VP\s*ĐU|VP\s*DU)\b\s*",
                "",
                RegexOptions.IgnoreCase
            );

            // Bỏ các dòng rác giữa header và trích yếu nếu có.
            var parts = s
                .Split(new[] { '\n' }, StringSplitOptions.RemoveEmptyEntries)
                .Select(x => Regex.Replace(x.Trim(), @"\s+", " "))
                .Where(x => !string.IsNullOrWhiteSpace(x))
                .Where(x => !Regex.IsMatch(FoldVietnameseKey(x), @"^(KHAN|THUONG KHAN|HOA TOC)$"))
                .Where(x => !Regex.IsMatch(FoldVietnameseKey(x), @"^(DANG BO|DANG UY|DANG CONG SAN|BINH TIEN NGAY)"))
                .ToList();

            s = string.Join(" ", parts);

            s = Regex.Replace(s, @"\s+", " ").Trim();
            s = s.Trim('-', '–', '—', '_', '=', '*', '.', ':', ';', ' ');

            if (string.IsNullOrWhiteSpace(s))
                return "";

            if (IsUiNoiseLine(s))
                return "";

            if (IsStandaloneDocTypeLine(s))
                return "";

            if (Regex.IsMatch(s, @"^(Kính\s*gửi|Kinh\s*gui|Căn\s+cứ|Can\s+cu)\b", RegexOptions.IgnoreCase))
                return "";

            return CleanFinalTrichYeuTitle(s);
        }

        // FIX_USER_REPORTED_11_CASES_METHOD
        // Chỉ sửa đúng 11 lỗi OCR/trích yếu đã được người dùng gửi ảnh và mô tả.
        // Không sửa pipeline. Không áp dụng cho văn bản khác nếu không khớp dấu hiệu.
        private void ApplyUserReportedCaseFixes(DocNumberInfo result, string rawOcrText)
        {
            if (result == null)
                return;

            string source =
                (rawOcrText ?? "") + " " +
                (result.TrichYeuNoiDung ?? "") + " " +
                (result.SoVanBan ?? "") + " " +
                (result.KyHieuVanBan ?? "");

            string key = NormalizeVietnameseLoose(source);
            key = Regex.Replace(key, @"\s+", " ").Trim();
            // FIX_USER_REPORTED_20260702_CASE_34_35_36_BEGIN
            // Chỉ sửa đúng 3 bản ghi người dùng vừa báo:
            // 36-KH/ĐU: lấy thiếu trích yếu.
            // 35-KH/ĐU: lấy thiếu trích yếu.
            // 34-KH/ĐU: lấy dư trích yếu.
            // Không đụng pipeline OCR/Selenium và không ảnh hưởng các bản ghi khác.

            if (
                KeyHasAll(key, "36", "kh/du", "suu tam", "hien tang", "bao tang dang cong san viet nam") ||
                KeyHasAll(key, "tuyen truyen van dong suu tam", "chuyen giao", "hien tang tai lieu", "phuong binh tien") ||
                KeyHasAll(key, "xay dung bao tang dang cong san viet nam", "dia ban phuong binh tien")
            )
            {
                SetReportedCase(
                    result,
                    "36",
                    "KH/ĐU",
                    "Kế hoạch",
                    "tuyên truyền, vận động sưu tầm, chuyển giao, hiến tặng tài liệu, tư liệu, hiện vật xây dựng Bảo tàng Đảng Cộng sản Việt Nam trên địa bàn phường Bình Tiên"
                );
                return;

        private bool KeyHasAll(string key, params string[] tokens)
        {
            if (string.IsNullOrWhiteSpace(key))
                return false;

            foreach (string token in tokens)
            {
                string t = NormalizeVietnameseLoose(token ?? "");
                t = Regex.Replace(t, @"\s+", " ").Trim();

                if (string.IsNullOrWhiteSpace(t))
                    continue;

                if (!key.Contains(t))
                    return false;
            }

            return true;
        }

        // FIX_BULK_VIETNAMESE_NORMALIZER_METHOD
        // Chuẩn hóa hàng loạt tiếng Việt sau OCR bằng file ngoài code:
        // ocr_vietnamese_normalization.tsv
        // Sau này gặp lỗi dấu/cụm từ mới chỉ cần thêm dòng sai<TAB>đúng vào file này.
        private string NormalizeVietnameseBulk(string text)
        {
            if (string.IsNullOrWhiteSpace(text))
                return "";

            string s = text.Normalize(System.Text.NormalizationForm.FormC);

            s = s.Replace("&amp;#43;", "+")
                 .Replace("&#43;", "+")
                 .Replace("&quot;", "\"")
                 .Replace("&amp;quot;", "\"")
                 .Replace("\u201C", "\"")
                 .Replace("\u201D", "\"")
                 .Replace("\u2018", "'")
                 .Replace("\u2019", "'")
                 .Replace("`", "'")
                 .Replace("´", "'");

            s = Regex.Replace(s, @"[ \t]+", " ");
            s = Regex.Replace(s, @"\n\s*\n+", "\n");

            foreach (var pair in LoadVietnameseBulkPairs())
            {
                string wrong = pair.Item1;
                string right = pair.Item2;

                if (string.IsNullOrWhiteSpace(wrong) || string.IsNullOrWhiteSpace(right))
                    continue;

                // Không sửa từ đơn quá ngắn để tránh phá chữ đúng.
                if (wrong.Trim().Length < 3)
                    continue;

                try
                {
                    string pattern =
                        @"(?<![\p{L}\p{N}])" +
                        Regex.Escape(wrong.Trim()).Replace(@"\ ", @"\s+") +
                        @"(?![\p{L}\p{N}])";

                    s = Regex.Replace(s, pattern, right.Trim(), RegexOptions.IgnoreCase);
                }
                catch
                {
                    // Một dòng từ điển lỗi thì bỏ qua dòng đó, không làm hỏng automation.
                }
            }

            // Chuẩn hóa ngày tháng không dấu:
            // ngay 23 thang 7 nam 2025 -> ngày 23 tháng 7 năm 2025
            s = Regex.Replace(
                s,
                @"(?<![\p{L}\p{N}])ngay\s+(\d{1,2})\s+thang\s+(\d{1,2})\s+nam\s+(\d{4})(?![\p{L}\p{N}])",
                "ngày $1 tháng $2 năm $3",
                RegexOptions.IgnoreCase
            );

            // Chuẩn hóa số/ký hiệu văn bản.
            s = Regex.Replace(
                s,
                @"(?<![\p{L}\p{N}])S[o0]\s*[:：]?\s*(\d{1,6})\s*[-–—]\s*([A-ZĐÂĂÊÔƠƯ0-9]{1,10})\s*/\s*([A-ZĐÂĂÊÔƠƯ0-9]{1,15})(?![\p{L}\p{N}])",
                "Số $1-$2/$3",
                RegexOptions.IgnoreCase
            );

            s = Regex.Replace(s, @"/DU\b", "/ĐU", RegexOptions.IgnoreCase);
            s = Regex.Replace(s, @"-DU\b", "-ĐU", RegexOptions.IgnoreCase);
            s = Regex.Replace(s, @"/D\s*U\b", "/ĐU", RegexOptions.IgnoreCase);
            s = Regex.Replace(s, @"/T[ƯU]\b", "/TW", RegexOptions.IgnoreCase);
            s = Regex.Replace(s, @"\bNQ\s*/\s*T[ƯU]\b", "NQ/TW", RegexOptions.IgnoreCase);
            s = Regex.Replace(s, @"\bCT\s*/\s*T[ƯU]\b", "CT/TW", RegexOptions.IgnoreCase);
            s = Regex.Replace(s, @"\bKH\s*/\s*T[ƯU]\b", "KH/TW", RegexOptions.IgnoreCase);

            // Dọn khoảng trắng nhưng giữ xuống dòng cho các hàm tách trích yếu.
            s = Regex.Replace(s, @"[ \t]+", " ");
            s = Regex.Replace(s, @"\n\s*\n+", "\n");

            return s.Trim();
        }

        private System.Collections.Generic.List<Tuple<string, string>> LoadVietnameseBulkPairs()
        {
            var result = new System.Collections.Generic.List<Tuple<string, string>>();

            string baseDir = AppDomain.CurrentDomain.BaseDirectory;

            string[] candidates =
            {
                Path.Combine(baseDir, "ocr_vietnamese_normalization.tsv"),
                Path.GetFullPath(Path.Combine(baseDir, @"..\ocr_vietnamese_normalization.tsv")),
                Path.GetFullPath(Path.Combine(baseDir, @"..\..\ocr_vietnamese_normalization.tsv")),
                Path.GetFullPath(Path.Combine(baseDir, @"..\..\..\ocr_vietnamese_normalization.tsv")),
                Path.GetFullPath(Path.Combine(baseDir, @"..\..\..\..\ocr_vietnamese_normalization.tsv"))
            };

            string file = candidates.FirstOrDefault(File.Exists);

            if (string.IsNullOrWhiteSpace(file))
                return result;

            try
            {
                foreach (string rawLine in File.ReadAllLines(file, System.Text.Encoding.UTF8))
                {
                    string line = rawLine == null ? "" : rawLine.Trim();

                    if (string.IsNullOrWhiteSpace(line))
                        continue;

                    if (line.StartsWith("#"))
                        continue;

                    string[] parts = line.Split(new[] { '\t' }, 2);

                    if (parts.Length < 2)
                        continue;

                    string wrong = parts[0].Trim();
                    string right = parts[1].Trim();

                    if (string.IsNullOrWhiteSpace(wrong) || string.IsNullOrWhiteSpace(right))
                        continue;

                    result.Add(Tuple.Create(wrong, right));
                }
            }
            catch
            {
                // Không được làm chết automation chỉ vì file từ điển lỗi.
            }

            return result;
        }

        private void ApplyVietnameseBulkToDocInfo(DocNumberInfo result)
        {
            if (result == null)

            result.SoVanBan = NormalizeVietnameseBulk(result.SoVanBan);
            result.KyHieuVanBan = NormalizeKyHieu(NormalizeVietnameseBulk(result.KyHieuVanBan));
            result.TenTheLoaiVanBan = NormalizeVietnameseBulk(result.TenTheLoaiVanBan);
            result.TrichYeuNoiDung = CleanFinalTrichYeuTitle(
                NormalizeVietnameseBulk(result.TrichYeuNoiDung)
            );
        }
        private string CleanNonCongVanTrichYeuFinal(string value)
        {
            if (string.IsNullOrWhiteSpace(value))
                return "";

            string s = value.Trim();

            s = FixCommonOcrCharacters(s);
            s = CutExtraBodyFromTrichYeuGeneric(s);
            s = FixCommonVietnameseOcrWords(s);
            s = CleanFinalTrichYeuTitle(s);

            return s;
        }
        private string ExtractVeTitleAnywhere(string rawText)
        {
            if (string.IsNullOrWhiteSpace(rawText))
                return "";

            var lines = rawText
                .Replace("\r", "\n")
                .Replace("\u00A0", " ")
                .Split(new[] { '\n' }, StringSplitOptions.RemoveEmptyEntries)
                .Select(x => CleanTrichYeuLine(x))
                .Where(x => !string.IsNullOrWhiteSpace(x))
                .ToList();

            for (int i = 0; i < lines.Count; i++)
            {
                string line = lines[i];
                int firstSoIndex = lines.FindIndex(x => Regex.IsMatch(x,
  @"^S[oố]\s*[:：]?\s*\d{1,6}\s*[-–—]\s*[A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]{1,10}(?:\s*[\/\-]\s*[A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]{1,15})+",
  RegexOptions.IgnoreCase));
                // Chỉ ưu tiên dòng "Về..." nằm gần Số/Ký hiệu văn bản ở phần đầu.
                // Tránh lấy nhầm dòng "Về..." hoặc nội dung bên dưới thân văn bản.
                if (firstSoIndex >= 0 && i > firstSoIndex + 8)
                    break;

                if (!Regex.IsMatch(line, @"^(Về|V/v|V\/v|Về việc)\b", RegexOptions.IgnoreCase))
                    continue;

                var parts = new System.Collections.Generic.List<string>();
                parts.Add(line);

                for (int j = i + 1; j < lines.Count && j <= i + 6; j++)
                {
                    string next = CleanTrichYeuLine(lines[j]);
                    string key = FoldVietnameseKey(next);

                    // Không dừng "THỰC HIỆN" nếu nó là dòng nối tiếp trích yếu:
                    // "... có thành tích trong"
                    // "thực hiện công tác..."
                    bool isThucHienContinuation =
                        Regex.IsMatch(key, @"^THUC HIEN\b", RegexOptions.IgnoreCase) &&
                        parts.Count > 0 &&
                        Regex.IsMatch(FoldVietnameseKey(parts[parts.Count - 1]), @"\b(TRONG|VE|CAC|NHIEM VU THEO)$", RegexOptions.IgnoreCase);

                    if (Regex.IsMatch(key,
                        @"^(KINH GUI|CAN CU|DE CO|NOI NHAN|TRAN TRONG|DANG UY|T/M|TM|NOI DUNG XAC MINH|NOI DUNG NHU SAU)\b",
                        RegexOptions.IgnoreCase))
                        break;

                    if (Regex.IsMatch(key, @"^THUC HIEN\b", RegexOptions.IgnoreCase) && !isThucHienContinuation)
                        break;

                    // Chỉ dừng dòng NGÀY khi nó là dòng mở đầu thân văn bản,
                    // không dừng dòng tiếp nối trích yếu: "ngày 07 tháng 11 năm 2025"
                    if (Regex.IsMatch(key, @"^NGAY\s+\d", RegexOptions.IgnoreCase) &&
                        !Regex.IsMatch(key, @"^NGAY\s+\d{1,2}\s+THANG\s+\d{1,2}\s+NAM\s+\d{4}$", RegexOptions.IgnoreCase))
                        break;

                    // FIX: nối nốt dòng "- 2030)" nếu OCR tách nhiệm kỳ.
                    if (Regex.IsMatch(next, @"^\d{4}\s*\)$", RegexOptions.IgnoreCase) &&
                        parts.Count > 0 &&
                        Regex.IsMatch(parts[parts.Count - 1], @"\(\s*nhiệm\s*kỳ\s+\d{4}$", RegexOptions.IgnoreCase))
                    {
                        parts[parts.Count - 1] = parts[parts.Count - 1] + " - " + next;
                        break;
                    }

                    if (Regex.IsMatch(key, @"^\d+\s*[\.\)]"))
                        break;

                    parts.Add(next);

                    if (Regex.IsMatch(key, @"\b(NOI CU TRU)\b", RegexOptions.IgnoreCase))
                        break;

                }

                string result = CleanCongVanTrichYeuFinal(string.Join(" ", parts));

                if (!string.IsNullOrWhiteSpace(result) && result.Length >= 6)
                    return result;
            }

            return "";
        }

        private string ExpandCongVanTrichYeuIfMissingNextLine(string currentTitle, string rawText)
        {
            if (string.IsNullOrWhiteSpace(currentTitle) || string.IsNullOrWhiteSpace(rawText))
                return currentTitle ?? "";

            string current = CleanFinalTrichYeuTitle(currentTitle);

            if (!Regex.IsMatch(current, @"^(Về|V/v|V\/v|Về việc)\b", RegexOptions.IgnoreCase))
                return current;

            var lines = rawText
                .Replace("\r", "\n")
                .Replace("\u00A0", " ")
                .Split(new[] { '\n' }, StringSplitOptions.RemoveEmptyEntries)
                .Select(x => Regex.Replace(x.Trim(), @"\s+", " "))
                .Where(x => !string.IsNullOrWhiteSpace(x))
                .ToList();

            string currentKey = FoldVietnameseKey(current);

            for (int i = 0; i < lines.Count; i++)
            {
                string line = CleanTrichYeuLine(lines[i]);
                string lineKey = FoldVietnameseKey(line);

                if (!lineKey.StartsWith(currentKey) && !currentKey.StartsWith(lineKey))
                    continue;

                var parts = new System.Collections.Generic.List<string>();
                parts.Add(line);

                for (int j = i + 1; j < lines.Count && j <= i + 4; j++)
                {
                    string next = CleanTrichYeuLine(lines[j]);

                    if (string.IsNullOrWhiteSpace(next))
                        break;

                    string nextKey = FoldVietnameseKey(next);

                    if (Regex.IsMatch(nextKey,
                          @"^(KINH GUI|CAN CU|THUC HIEN|DE CO|NOI NHAN|TRAN TRONG|DANG UY|T/M|TM|NAY[,\s]|NOI DUNG NHU SAU|KHAN|THUONG KHAN|HOA TOC)\b",
                          RegexOptions.IgnoreCase))
                        break;

                    if (Regex.IsMatch(nextKey, @"^\d+\s*[\.\)]"))
                        break;

                    if (Regex.IsMatch(nextKey, @"^[IVX]+\s*[\.\)]"))
                        break;

                    if (Regex.IsMatch(nextKey, @"^[-–—=_*. ]+$"))
                        break;

                    if (Regex.IsMatch(nextKey, @"^\d+\s*[\.\)]"))
                        break;

                    parts.Add(next);

                    if (next.EndsWith("."))
                        break;
                }

                string expanded = CleanFinalTrichYeuTitle(string.Join(" ", parts));

                if (expanded.Length > current.Length)
                    return expanded;

                return current;
            }

            return current;
        }

        private string CleanCongVanTrichYeuFinal(string value)
        {

            if (string.IsNullOrWhiteSpace(value))
                return "";

            string s = value.Trim();

            // Sửa HTML entity
            s = s.Replace("&quot;", "\"")
                 .Replace("&amp;quot;", "\"")
                 .Replace("\u201C", "\"")
                 .Replace("\u201D", "\"");

            // Xóa "KHẨN" / "THƯỢNG KHẨN" / "HỎA TỐC" nếu dính vào đầu hoặc cuối
            s = Regex.Replace(s,
@"^\s*(KHẨN|KHẮN|KHẤN|KHAN|KHANH|THƯỢNG\s*KHẨN|THUONG\s*KHAN|HỎA\s*TỐC|HOA\s*TOC)\s*", "", RegexOptions.IgnoreCase).Trim();
            s = Regex.Replace(s,
@"\s*(KHẨN|KHẮN|KHẤN|KHAN|KHANH|THƯỢNG\s*KHẨN|THUONG\s*KHAN|HỎA\s*TỐC|HOA\s*TOC)\s*$", "", RegexOptions.IgnoreCase).Trim();

            // Cắt lỗi dính đầu: "Số 70-CV/ĐU Về việc..."
            s = Regex.Replace(
                s,
                @"^\s*S[oố]\s*[:：]?\s*\d{1,6}\s*[-–—]\s*[A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]+(?:\s*[\/\-]\s*[A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]+)+\s*",
                "", RegexOptions.IgnoreCase);

            // Nếu bị dính "Công văn số 29-CV/ĐU...", chỉ lấy phần từ "Về" trở đi
            Match mVe = Regex.Match(s,
                @"(?<title>(?:Về|V/v|V\/v|Về việc|Báo cáo về)\b.+)$",
                RegexOptions.IgnoreCase);
            if (mVe.Success)
                s = mVe.Groups["title"].Value;
            // Cắt KHẨN/KHẮN nếu OCR dính sau trích yếu rồi kéo theo "Kính gửi"
            s = Regex.Replace(
                s,
                @"\s+(KHẨN|KHẮN|KHẤN|KHAN|KHANH|THƯỢNG\s*KHẨN|THUONG\s*KHAN|HỎA\s*TỐC|HOA\s*TOC)\b.*$",
                "",
                RegexOptions.IgnoreCase
            ).Trim();
            // SỬA: Không cắt theo "Nay[,\s]" ở giữa câu,
            // vì sẽ cắt nhầm cụm hợp lệ "hiện nay".
            // Cắt phần thân bị dính sau tên đồng chí.
            // Ví dụ: ", chức vụ: Trung đội trưởng, cấp bậc..."
            s = Regex.Replace(
                s,
                @"\s*,?\s*(?=chức\s+vụ|chuc\s+vu|cấp\s+bậc|cap\s+bac|cấp\s+bộc|cap\s+boc).*$",
                "",
                RegexOptions.IgnoreCase
            ).Trim();

            // Cắt phần nghề nghiệp/đơn vị công tác bị dính sau tên đồng chí.
            // Ví dụ: "Dược sĩ CKI, Phòng Kế hoạch - Tổng hợp..."
            s = Regex.Replace(
                s,
                @"\s*[\.,;]?\s*(?=Dược\s+sĩ|Duoc\s+si|Bác\s+sĩ|Bac\s+si|Thạc\s+sĩ|Thac\s+si|Tiến\s+sĩ|Tien\s+si|Phòng\s+Kế\s+hoạch|Phong\s+Ke\s+hoach|Trung\s+tâm\s+Pháp\s+y|Trung\s+tam\s+Phap\s+y).*$",
                "",
                RegexOptions.IgnoreCase
            ).Trim();

            // Cắt phần thân xác minh nếu OCR dính tiếp sau trích yếu.
            s = Regex.Replace(
                s,
                @"\s+Đảng\s+ủy\s+phường\s+Bình\s+Tiên\s+xác\s+minh\s+như\s+sau.*$",
                "",
                RegexOptions.IgnoreCase
            ).Trim();

            // Cắt phần thân thẩm tra/xác minh bị dính sau trích yếu.
            // Ví dụ: "DNội dung thẩm tra như sau..." hoặc "Nội dung thẩm tra như sau..."
            s = Regex.Replace(
                s,
                @"\s*D?\s*(?=Nội\s+dung\s+thẩm\s+tra|Noi\s+dung\s+tham\s+tra|Nội\s+dung\s+xác\s+minh|Noi\s+dung\s+xac\s+minh).*$",
                "",
                RegexOptions.IgnoreCase
            ).Trim();

            // Cắt phần đề nghị bị dính sau trích yếu.
            s = Regex.Replace(
                s,
                @"\s+(?=Đề\s+nghị\s+Ban|De\s+nghi\s+Ban).*$",
                "",
                RegexOptions.IgnoreCase
            ).Trim();

            s = Regex.Replace(s,
  @"\s*(?=Kính\s*gửi|Kinh\s*gửi|Kính\s*gui|Kinh\s*gui|Căn\s+cứ|Can\s+cu|Nơi\s+nhận|Noi\s+nhan|Trân\s+trọng|Tran\s+trong|Nội\s+dung\s+xác\s+minh|Noi\s+dung\s+xac\s+minh|Nội\s+dung\s+như\s+sau|Noi\s+dung\s+nhu\s+sau|\d+\s*[\.\)]\s+[A-ZĐÂĂÊÔƠƯÀ-Ỹ]).*$",
  "",
  RegexOptions.IgnoreCase).Trim();

            s = Regex.Replace(s, @"\s+", " ").Trim();
            s = s.Trim('-', '–', '—', '_', '=', '*', ':', ';', ' ');

            return s.Trim();
        }

        private string OcrLocalFromPdfViewer()
        {
            Bitmap pageImage = null;
            Bitmap headerImage = null;
            Bitmap scaledImage = null;

            try
            {
                pageImage = CapturePdfPageFromBrowserScreenshot();

                // Cắt vùng đầu trang: nơi có Số, Ký hiệu, KẾ HOẠCH/QUYẾT ĐỊNH/BÁO CÁO và trích yếu.
                headerImage = CropHeaderArea(pageImage);

                // Phóng to ảnh để Tesseract đọc rõ hơn.
                scaledImage = ScaleBitmap(headerImage, 3);

                string imgPath = Path.Combine(
                    Path.GetTempPath(),
                    "pdf_header_ocr_" + Guid.NewGuid().ToString("N") + ".png"
                );

                scaledImage.Save(imgPath, System.Drawing.Imaging.ImageFormat.Png);

                AddLog("OCR local", "Đang làm", "Đã lưu ảnh vùng đầu PDF: " + imgPath);

                string text = RunTesseractOcr(imgPath);

                return text;
            }
            finally
            {
                if (pageImage != null) pageImage.Dispose();
                if (headerImage != null) headerImage.Dispose();
                if (scaledImage != null) scaledImage.Dispose();
            }
        }

                        full,
                        new Rectangle(0, 0, cropped.Width, cropped.Height),
                        cropRect,
                        GraphicsUnit.Pixel
                    );
                }

                return cropped;
            }

        private Bitmap CropHeaderArea(Bitmap source)
        {
            // Lấy khoảng 50% đầu trang đang hiển thị.
            // Nếu cần lấy thêm trích yếu dài, tăng 0.50 lên 0.60.
            int x = 0;
            int y = 0;
            int width = source.Width;
            int height = (int)(source.Height * 0.60);

            Rectangle cropRect = new Rectangle(x, y, width, height);

            Bitmap cropped = new Bitmap(cropRect.Width, cropRect.Height);

            using (Graphics g = Graphics.FromImage(cropped))
            {
                g.DrawImage(
                    source,
                    new Rectangle(0, 0, cropped.Width, cropped.Height),
                    cropRect,
                    GraphicsUnit.Pixel
                );
            }

            return cropped;
        }

        private string NormalizeGenericTrichYeuAfterLocalOcr(string currentTitle, string rawOcrText)
        {
            if (string.IsNullOrWhiteSpace(currentTitle))
                return "";

            string s = currentTitle.Trim();

            s = FixCommonOcrCharacters(s);
            s = CutExtraBodyFromTrichYeuGeneric(s);
            s = FixCommonVietnameseOcrWords(s);
            s = CleanFinalTrichYeuTitle(s);

            return s;
        }
        private string FixCommonOcrCharacters(string value)
        {
            if (string.IsNullOrWhiteSpace(value))
                return "";

            string s = value.Trim();

            // Dọn ký tự OCR lạ.
            s = s.Replace("`", "'")
                 .Replace("´", "'")
                 .Replace("’", "'")
                 .Replace("‘", "'")
                 .Replace("“", "\"")
                 .Replace("”", "\"")
                 .Replace("§", "8");

            // Chuẩn hóa TW trong ký hiệu.
            // Ví dụ: CT/TƯ, KH/TƯ -> CT/TW, KH/TW
            s = Regex.Replace(
                s,
                @"/\s*T[ƯU]\b",
                "/TW",
                RegexOptions.IgnoreCase
            );

            // Chuẩn hóa khoảng trắng quanh mã văn bản.
            // Ví dụ: 28 - CT / TW -> 28-CT/TW
            s = Regex.Replace(
                s,
                @"\b(\d{1,6})\s*[-–—]\s*([A-ZĐÂĂÊÔƠƯ]{1,10})\s*/\s*([A-ZĐÂĂÊÔƠƯ]{1,10})\b",
                "$1-$2/$3",
                RegexOptions.IgnoreCase
            );

            // Dọn dấu nháy dư trước dấu chấm phẩy.
            s = Regex.Replace(s, @"['`´’‘]\s*;", ";");

            s = Regex.Replace(s, @"\s+", " ").Trim();

            return s;
        }

        private string CutExtraBodyFromTrichYeuGeneric(string value)
        {
            if (string.IsNullOrWhiteSpace(value))
                return "";

            string s = value.Trim();
            s = Regex.Replace(s, @"\s+", " ").Trim();

            /*
             * CẤM cắt theo dấu phẩy.
             * Lỗi cũ: thấy "tiếp tục thực hiện Chỉ thị số 28-CT/TW ... rà soát,"
             * rồi cắt từ chữ "thực hiện", làm kết quả chỉ còn "tiếp tục".
             *
             * Chỉ cắt "Thực hiện ..." khi nó thật sự là câu mở đầu phần thân:
             * - có dấu ; hoặc :
             * - hoặc phía sau có "như sau"
             * - hoặc phía sau có "xây dựng / ban hành / triển khai"
             * - và KHÔNG được cắt nếu ngay trước nó là "tiếp tục".
             */
            s = CutAtValidBodyThucHienMarker(s);

            string[] cutPatterns =
            {
        // Dòng thân dạng: Ban Thường vụ ... xây dựng ... như sau
        @"\s+Ban\s+Thường\s+vụ\s+.{0,180}\s+(xây\s+dựng|ban\s+hành|triển\s+khai)\s+.{0,180}\s+như\s+sau.*$",

        // Các mục thân văn bản.

        @"\s+[IVX]+\.\s*MUC\s+DICH.*$",
        @"\s+[IVX]+\.\s+MỤC\s+ĐÍCH,\s*YÊU\s+CẦU.*$",
        @"\s+[IVX]+\.\s+MUC\s+DICH,\s*YEU\s+CAU.*$",
        @"\s+\d+\.\s*Mục\s+đích.*$",
        @"\s+\d+\.\s*Muc\s+dich.*$",
        @"\s+\d+\.\s+Yêu\s+cầu.*$",

        @"\s+PHẦN\s+THỨ\s+.*$",
        @"\s+PHAN\s+THU\s+.*$"

            {
                s = Regex.Replace(s, pattern, "", RegexOptions.IgnoreCase);
            }

            s = Regex.Replace(s, @"\s+", " ").Trim();
            s = CleanFinalTrichYeuTitle(s);

            return s;

        private string CutAtValidBodyThucHienMarker(string value)
        {
            if (string.IsNullOrWhiteSpace(value))
                return "";

            string s = value.Trim();

            string[] markerPatterns =
            {
        @"\bThực\s+hiện\s+(Chỉ\s+thị|Công\s+văn|Kế\s+hoạch|Quyết\s+định|Nghị\s+quyết|Thông\s+báo)\s+s[oố]\s+\d{1,6}\s*[-–—]",
        @"\bThuc\s+hien\s+(Chi\s+thi|Cong\s+van|Ke\s+hoach|Quyet\s+dinh|Nghi\s+quyet|Thong\s+bao)\s+so\s+\d{1,6}\s*[-–—]"
    };

            int cutIndex = -1;

            foreach (string pattern in markerPatterns)
            {
                MatchCollection matches = Regex.Matches(s, pattern, RegexOptions.IgnoreCase);

                foreach (Match m in matches)
                {
                    if (!m.Success)
                        continue;

                    string before = s.Substring(0, m.Index).Trim();
                    string tail = s.Substring(m.Index).Trim();

                    // Nếu marker nằm ở đầu chuỗi thì không cắt,
                    // vì có nhiều trích yếu hợp lệ bắt đầu bằng "Thực hiện..."
                    if (string.IsNullOrWhiteSpace(before))
                        continue;

                    // Bảo vệ trường hợp hợp lệ:
                    // "tiếp tục thực hiện Chỉ thị số..."
                    if (IsProtectedThucHienInsideTitle(before))
                        continue;

                    string tailKey = FoldVietnameseKey(tail);

                    bool looksLikeBody =
                        tail.Contains(";") ||
                        tail.Contains(":") ||
                        tailKey.Contains(" NHU SAU") ||
                        tailKey.Contains(" XAY DUNG ") ||
                        tailKey.Contains(" BAN HANH ") ||
                        tailKey.Contains(" TRIEN KHAI ");

                    if (!looksLikeBody)
                        continue;

                    if (cutIndex < 0 || m.Index < cutIndex)
                        cutIndex = m.Index;
                }
            }

            if (cutIndex >= 0)
                return s.Substring(0, cutIndex).Trim();

            return s;
        }

        private bool IsProtectedThucHienInsideTitle(string before)
        {
            if (string.IsNullOrWhiteSpace(before))
                return true;

            string key = FoldVietnameseKey(before);
            key = Regex.Replace(key, @"\s+", " ").Trim();

            // Không được cắt các cụm trích yếu hợp lệ:
            // "tiếp tục thực hiện..."
            // "về việc thực hiện..."
            // "nhằm thực hiện..."
            // "để thực hiện..."
            return Regex.IsMatch(
                key,
                @"(TIEP\s+TUC|VE|VE\s+VIEC|NHAM|DE|TO\s+CHUC|TRIEN\s+KHAI|PHOI\s+HOP)\s*$",
                RegexOptions.IgnoreCase
            );
        }

        private Bitmap ScaleBitmap(Bitmap source, int scale)
        {
            int width = source.Width * scale;
            int height = source.Height * scale;

            Bitmap result = new Bitmap(width, height);

            using (Graphics g = Graphics.FromImage(result))
            {
                g.InterpolationMode = System.Drawing.Drawing2D.InterpolationMode.HighQualityBicubic;
                g.SmoothingMode = System.Drawing.Drawing2D.SmoothingMode.HighQuality;
                g.PixelOffsetMode = System.Drawing.Drawing2D.PixelOffsetMode.HighQuality;
                g.DrawImage(source, new Rectangle(0, 0, width, height));
            }

            return result;
        }

        private string RunTesseractOcr(string imagePath)
        {
            string tesseractExe = @"C:\Program Files\Tesseract-OCR\tesseract.exe";

            if (!File.Exists(tesseractExe))
                throw new Exception("Chưa tìm thấy Tesseract OCR tại: " + tesseractExe);

            string outputBase = Path.Combine(
                Path.GetTempPath(),
                "ocr_pdf_result_" + Guid.NewGuid().ToString("N")
            );

            var psi = new System.Diagnostics.ProcessStartInfo();
            psi.FileName = tesseractExe;
            psi.Arguments =
                "\"" + imagePath + "\" " +
                "\"" + outputBase + "\" " +
                "-l vie+eng --oem 1 --psm 6 -c preserve_interword_spaces=1";

            psi.CreateNoWindow = true;
            psi.UseShellExecute = false;
            psi.RedirectStandardOutput = true;
            psi.RedirectStandardError = true;

            string error = "";

            using (var p = System.Diagnostics.Process.Start(psi))
            {
                error = p.StandardError.ReadToEnd();
                p.WaitForExit();
            }

            string txtPath = outputBase + ".txt";

            if (!File.Exists(txtPath))
                throw new Exception("Tesseract không tạo file text. Lỗi: " + error);

            string text = File.ReadAllText(txtPath, System.Text.Encoding.UTF8);

            text = text.Replace("\r", "\n");
            text = Regex.Replace(text, @"[ \t]+", " ");
            text = Regex.Replace(text, @"\n\s*\n+", "\n");

            // FIX_BULK_VIETNAMESE_NORMALIZER_TESSERACT_CALL
            text = NormalizeVietnameseBulk(text);

            return text.Trim();
        }

        /// <summary>
        /// Tìm trích yếu trực tiếp bằng regex trên toàn bộ text OCR.
        /// Đáng tin hơn cách line-by-line vì không phụ thuộc vào định dạng newline.
        /// </summary>
        private string ExtractTrichYeuByDirectSearch(string rawText)
        {
            if (string.IsNullOrWhiteSpace(rawText)) return "";

            // Chuẩn hóa xuống dòng
            string text = rawText.Replace("\r\n", "\n").Replace("\r", "\n");
            // BỔ SUNG: Công văn có trích yếu bắt đầu bằng "Về..." ngay dưới dòng Số/Ký hiệu.
            // Ví dụ:
            // Số:74-CV/ĐU
            // Về gửi tài liệu và xin ý kiến
            // Kính gửi: ...
            Match congVanTitle = Regex.Match(text,
            @"(?im)^\s*(?:S[oố]\s*[:：]?\s*\d{1,6}\s*[-–—]\s*[A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]+(?:\s*[\/\-]\s*[A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]+)+)\s*\n+" +
            @"(?<trichyeu>\s*(?:Về|V/v|V\/v|Về việc)\b[\s\S]*?)" +
@"(?=^\s*(?:Kính\s*gửi|Kinh\s*gui|Căn\s+cứ|Can\s*cu|Thực\s*hiện|Thuc\s*hien|Để\s+có|De\s+co|Đảng\s+ủy|Dang\s+uy|Nơi\s+nhận|Noi\s+nhan|Trân\s+trọng|Tran\s+trong)\b)",
            RegexOptions.IgnoreCase | RegexOptions.Multiline);

            if (congVanTitle.Success)
            {
                string cvTrichYeu = congVanTitle.Groups["trichyeu"].Value;
                cvTrichYeu = Regex.Replace(cvTrichYeu, @"\s+", " ").Trim();
                // BỔ SUNG: nếu regex vẫn lấy thiếu dòng tiếp theo của trích yếu "Về...",
                // thì đọc lại theo từng dòng và nối đến trước "Kính gửi".
                var cvLines = text
                    .Split(new[] { '\n' }, StringSplitOptions.RemoveEmptyEntries)
                    .Select(x => Regex.Replace(x.Trim(), @"\s+", " "))
                    .Where(x => !string.IsNullOrWhiteSpace(x))
                    .ToList();

                for (int i = 0; i < cvLines.Count; i++)
                {
                    if (!Regex.IsMatch(cvLines[i],
                        @"^S[oố]\s*[:：]?\s*\d{1,6}\s*[-–—]\s*[A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]+(?:\s*[\/\-]\s*[A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]+)+",
                        RegexOptions.IgnoreCase))
                        continue;

                    var cvParts = new System.Collections.Generic.List<string>();

                    for (int j = i + 1; j < cvLines.Count && j <= i + 8; j++)
                    {
                        string line = cvLines[j];

                        // Bỏ qua KHẨN / HỎA TỐC nằm giữa header và trích yếu
                        if (Regex.IsMatch(line.Trim(),
@"\s*(KHẨN|KHẮN|KHẤN|KHAN|KHANH|THƯỢNG\s*KHẨN|THUONG\s*KHAN|HỎA\s*TỐC|HOA\s*TOC)\s*$", RegexOptions.IgnoreCase))
                            continue;

                        if (Regex.IsMatch(line,
                            @"^(Kính\s*gửi|Kinh\s*gui|Căn\s+cứ|Can\s*cu|Thực\s*hiện|Thuc\s*hien|Để\s+có|De\s+co|Nơi\s+nhận|Noi\s+nhan|Trân\s+trọng|Tran\s+trong|Nay[,\s]|Nội\s+dung\s+như\s+sau)\b",
                            RegexOptions.IgnoreCase))
                            break;

                        // Dừng trước mục đánh số: 1. / 2. / I. / II.
                        if (Regex.IsMatch(line, @"^\d+\s*[\.\)]\s+", RegexOptions.IgnoreCase))
                            break;
                        if (Regex.IsMatch(line, @"^[IVXLCDM]+\s*[\.\)]\s+", RegexOptions.IgnoreCase))
                            break;

                        if (cvParts.Count == 0 &&
                            !Regex.IsMatch(line, @"^(Về|V/v|V\/v|Về việc|Báo cáo về)\b", RegexOptions.IgnoreCase))
                            continue;
                        // Kiểm tra stop-line lần cuối trước khi thêm vào
                        string lineCheck = Regex.Replace(line.ToLowerInvariant(), @"\s+", " ").Trim();
                        if (lineCheck.StartsWith("kính gửi") || lineCheck.StartsWith("kinh gui"))
                            break;

                        cvParts.Add(line);

                        if (line.TrimEnd().EndsWith("."))
                            break;
                    }

                    string lineByLineTitle = Regex.Replace(string.Join(" ", cvParts), @"\s+", " ").Trim();

                    if (lineByLineTitle.Length > cvTrichYeu.Length)
                        cvTrichYeu = lineByLineTitle;

                    break;
                }

                // BỔ SUNG: OCR mất chữ "Số ...-CV/ĐU", chỉ còn số rời như "228",
                // nhưng trích yếu vẫn bắt đầu bằng "Về...".
                Match veTitleWithoutSoKyHieu = Regex.Match(text,
                 @"(?im)^\s*(?<trichyeu>(?:Về|V/v|V\/v|Về việc|Báo cáo về)\b[\s\S]*?)" +
                 @"(?=^\s*(?:Kính\s*gửi|Kinh\s*gui|Căn\s+cứ|Can\s*cu|Thực\s*hiện|Thuc\s*hien|Ngày\s+\d|Ngay\s+\d|Nay[,\s]|Nội\s+dung\s+như\s+sau|Đảng\s+ủy|Dang\s+uy|Nơi\s+nhận|Noi\s+nhan|Trân\s+trọng|Tran\s+trong|\d+\s*[\.\)]|[IVXLCDM]+\s*[\.\)])\b)",
                 RegexOptions.IgnoreCase | RegexOptions.Multiline);

                if (veTitleWithoutSoKyHieu.Success)
                {
                    string veTrichYeu = veTitleWithoutSoKyHieu.Groups["trichyeu"].Value;
                    veTrichYeu = Regex.Replace(veTrichYeu, @"\s+", " ").Trim();
                    veTrichYeu = veTrichYeu.Trim('-', '–', '—', '_', '=', '*', '.', ':', ';', ' ');

                    if (!string.IsNullOrWhiteSpace(veTrichYeu) &&
                        veTrichYeu.Length >= 6 &&
                        !IsUiNoiseLine(veTrichYeu) &&
                        !IsStandaloneDocTypeLine(veTrichYeu))
                    {
                        return veTrichYeu;
                    }
                }
                // Cắt stop-line nếu dính vào cuối — chỉ cắt khi stop-word ở ĐẦU DÒNG mới
                // để không cắt nhầm "hiện nay", "thực hiện" nằm trong trích yếu
                {
                    var cvTrichYeuLines2 = cvTrichYeu.Split('\n');
                    var kept = new System.Collections.Generic.List<string>();
                    foreach (var ln2 in cvTrichYeuLines2)
                    {
                        string ln2t = ln2.Trim();
                        if (Regex.IsMatch(ln2t,
                            @"^(Kính\s*gửi|Kinh\s*gui|Căn\s+cứ|Can\s*cu|Nơi\s+nhận|Noi\s+nhan|Trân\s+trọng|Tran\s+trong|Nay[,\s]|Nội\s+dung\s+như\s+sau|\d+\s*[\.\)]\s)",
                            RegexOptions.IgnoreCase))
                            break;
                        kept.Add(ln2t);
                    }
                    cvTrichYeu = string.Join(" ", kept).Trim();
                }

                // Xóa KHẨN nếu dính vào đầu hoặc cuối cvTrichYeu
                cvTrichYeu = Regex.Replace(cvTrichYeu,
                    @"^\s*(KHẨN|THƯỢNG\s*KHẨN|HỎA\s*TỐC)\s*",
                    "", RegexOptions.IgnoreCase).Trim();
                cvTrichYeu = Regex.Replace(cvTrichYeu,
                    @"\s*(KHẨN|THƯỢNG\s*KHẨN|HỎA\s*TỐC)\s*$",
                    "", RegexOptions.IgnoreCase).Trim();

                cvTrichYeu = cvTrichYeu.Trim('-', '–', '—', '_', '=', '*', '.', ':', ';', ' ');

                if (!string.IsNullOrWhiteSpace(cvTrichYeu) &&
                    cvTrichYeu.Length >= 6 &&
                    !IsUiNoiseLine(cvTrichYeu) &&
                    !IsStandaloneDocTypeLine(cvTrichYeu))
                {
                    return cvTrichYeu;
                }
            }

            // Tìm mẫu: [loại văn bản] + newline + [trích yếu 1 hoặc nhiều dòng] + dấu dừng
            // RegexOptions.Singleline để . khớp cả ký tự newline → xử lý được trích yếu nhiều dòng
            Match m = Regex.Match(text,
@"(?im)^\s*(?:QUYẾT\s*ĐỊNH|QUYẾT\s*ĐINH|QUYET\s*DINH|BÁO\s*CÁO|BAO\s*CAO|KẾ\s*HOẠCH|KE\s*HOACH|THÔNG\s*BÁO|THONG\s*BAO|TỜ\s*TRÌNH|TO\s*TRINH|NGHỊ\s*QUYẾT|NGHI\s*QUYET|CÔNG\s*VĂN|CONG\s*VAN|CHƯƠNG\s*TRÌNH|CHUONG\s*TRINH|QUY\s*CHẾ|QUY\s*CHE|HƯỚNG\s*DẪN|HUONG\s*DAN|BIÊN\s*BẢN|BIEN\s*BAN|CHỈ\s*THỊ|CHI\s*THI)\s*\n+" + @"(?<trichyeu>[\s\S]*?)" +
@"(?=^\s*[-–—]?\s*(?:Kính\s*gửi|Kinh\s*gui|Căn\s*cứ|Can\s*cu|Thực\s*hiện|Thuc\s*hien|Ban\s*Thường\s*vụ|Ban\s*Thuong\s*vu|Ban\s*Chấp\s*hành|Ban\s*Chap\s*hanh|Nay[,\s]|\d+\s*[\.\)]|[IVXLCDM]+\s*[\.\)])\b)", RegexOptions.IgnoreCase | RegexOptions.Multiline);

            if (!m.Success) return "";

            string trichYeu = m.Groups["trichyeu"].Value;

            // Trích yếu nhiều dòng: gộp lại thành 1 dòng
            trichYeu = Regex.Replace(trichYeu, @"\s+", " ").Trim();
            // FIX: Cắt phần thân bắt đầu bằng mục đánh số như "1. Công tác..."
            trichYeu = Regex.Split(
                trichYeu,
                @"\s+(?=\d+\s*[\.\)]\s+[A-ZĐÂĂÊÔƠƯÀ-Ỹ])",
                RegexOptions.IgnoreCase
            )[0].Trim();
            // Cắt nếu có dấu chấm/gạch kéo dài
            trichYeu = Regex.Split(trichYeu, @"\.{3,}|…+|[-–—_]{3,}|={3,}")[0];
            trichYeu = trichYeu.Trim('-', '–', '—', '_', '=', '*', '.', ':', ';', ' ');

            if (string.IsNullOrWhiteSpace(trichYeu) || trichYeu.Length < 6) return "";
            if (IsUiNoiseLine(trichYeu)) return "";
            if (IsStandaloneDocTypeLine(trichYeu)) return "";

            return trichYeu;
        }

        private string EnsureDocumentTextWhenOcrPopupFailed(string text)
        {
            text = text ?? "";

            if (LooksLikeDocumentText(text))
                return text;

            string viewerText = TryReadPdfViewerTextLayerFallback();

            if (LooksLikeDocumentText(viewerText))
            {
                AddLog("OCR fallback", "OK", "OCR toàn văn không hợp lệ, đã lấy textLayer PDF bên phải.");
                return viewerText;
            }

            return text;
        }

                            .filter(visible)
                            .slice(0, 2);

                        for (let layer of layers) {
                            let t = clean(layer.innerText || layer.textContent || '');
                            if (t.length > 0) parts.push(t);
                        }
                    }

                    return clean(parts.join('\n'));
                ");

                return value == null ? "" : value.ToString();
            }
            catch
            {
                return "";
            }
        }

        private string FixTrichYeuNoiDungByTitleRules(string currentValue, string rawText)
        {
            string current = CleanFinalTrichYeuTitle(currentValue);
            string raw = rawText ?? "";

            string strictTitle = TryExtractTitleBlockAfterDocumentType(raw);

            if (IsBetterTrichYeuTitle(strictTitle, current))
                current = strictTitle;

            current = TrimWrongBodyPartFromTitle(current);
            current = CleanFinalTrichYeuTitle(current);

            return current;
        }

        private string TryExtractTitleBlockAfterDocumentType(string rawText)
        {
            if (string.IsNullOrWhiteSpace(rawText))
                return "";

            var lines = rawText
                .Replace("\r", "\n")
                .Replace("\u00A0", " ")
                .Split(new[] { '\n' }, StringSplitOptions.RemoveEmptyEntries)
                .Select(x => CleanTitleLineForExtract(x))
                .Where(x => !string.IsNullOrWhiteSpace(x))
                .Where(x => !IsUiNoiseLine(x))
                .ToList();

            if (lines.Count == 0)
                return "";

            int maxScan = Math.Min(lines.Count, 80);

            for (int i = 0; i < maxScan; i++)
            {
                string line = lines[i];

                if (!IsDocumentTypeHeaderForTitleExtract(line))
                    continue;

                var parts = new System.Collections.Generic.List<string>();

                string rest = GetTitleRemainderFromDocumentTypeLine(line);

                if (!string.IsNullOrWhiteSpace(rest))
                    parts.Add(rest);

                for (int j = i + 1; j < lines.Count && parts.Count < 8; j++)
                {
                    string candidate = CleanTitleLineForExtract(lines[j]);

                    if (IsSkipNoiseLineForTitleExtract(candidate))
                        continue;

                    // Cho phép dòng đầu dưới KẾ HOẠCH/QUYẾT ĐỊNH/BÁO CÁO là:
                    // "tiếp tục thực hiện Chỉ thị..."
                    bool isFirstTitleLineThucHienChiThi =
                        parts.Count == 0 &&
                        Regex.IsMatch(FoldVietnameseKey(candidate),
                            @"^(TIEP\s+TUC\s+)?THUC\s+HIEN\s+CHI\s+THI\b",
                            RegexOptions.IgnoreCase);

                    if (IsStopLineForTitleBlock(candidate, parts.Count) && !isFirstTitleLineThucHienChiThi)
                        break;

                    if (!IsPossibleTitleLineForExtract(candidate))
                        continue;

                    parts.Add(candidate);
                }

                string title = CleanFinalTrichYeuTitle(string.Join(" ", parts));
                title = TrimWrongBodyPartFromTitle(title);

                if (!string.IsNullOrWhiteSpace(title) && title.Length >= 8)
                    return title;
            }

            return "";
        }

        private string CleanTitleLineForExtract(string line)
        {
            if (line == null)
                return "";

            string s = Regex.Replace(line, @"\s+", " ").Trim();

            s = Regex.Split(s, @"(\.{3,}|…+|[-–—_]{4,}|={3,})")[0];
            s = s.Trim('-', '–', '—', '_', '=', '*', '.', ':', ';', ' ');

            return s.Trim();
        }

        private string GetTitleRemainderFromDocumentTypeLine(string line)
        {
            if (string.IsNullOrWhiteSpace(line))
                return "";

            Match m = Regex.Match(
                line,
                @"^\s*(BÁO\s*CÁO|BAO\s*CAO|QUYẾT\s*ĐỊNH|QUYET\s*DINH|KẾ\s*HOẠCH|KE\s*HOACH|CÔNG\s*VĂN|CONG\s*VAN|CHỈ\s*THỊ|CHI\s*THI|TỜ\s*TRÌNH|TO\s*TRINH|THÔNG\s*BÁO|THONG\s*BAO|BIÊN\s*BẢN|BIEN\s*BAN)\s*[:\-–—]?\s*(?<rest>.+?)\s*$",
                RegexOptions.IgnoreCase);

            if (!m.Success)
                return "";

            string rest = CleanTitleLineForExtract(m.Groups["rest"].Value);

            if (string.IsNullOrWhiteSpace(rest))
                return "";

            string key = FoldVietnameseKey(rest);

            // Không lấy các dòng kiểu: BÁO CÁO NGÀY 27 THÁNG 11 NĂM 2025
            if (Regex.IsMatch(key, @"^NGAY\s+\d{1,2}\s+THANG\s+\d{1,2}\s+NAM\s+\d{4}") ||
                Regex.IsMatch(key, @"^\d{1,2}\s+THANG\s+\d{1,2}\s+NAM\s+\d{4}"))
                return "";

            if (IsStopLineForTitleBlock(rest, 0))
                return "";

            if (rest.Length < 8)
                return "";

            return rest;
        }

        private bool IsDocumentTypeHeaderForTitleExtract(string line)
        {
            if (string.IsNullOrWhiteSpace(line))
                return false;

            string key = FoldVietnameseKey(line);

            return Regex.IsMatch(
                key,
                @"^(BAO\s*CAO|QUYET\s*DINH|KE\s*HOACH|CONG\s*VAN|CHI\s*THI|TO\s*TRINH|THONG\s*BAO|BIEN\s*BAN)\b");
        }

        private bool IsSkipNoiseLineForTitleExtract(string line)
        {
            if (string.IsNullOrWhiteSpace(line))
                return true;

            string key = FoldVietnameseKey(line);

            if (Regex.IsMatch(key, @"^[\-\–\—=_\*\s]+$"))
                return true;

            if (key == "*")
                return true;

            if (key.Contains("DANG BO THANH PHO HO CHI MINH"))
                return true;

            if (key.Contains("DANG CONG SAN VIET NAM"))
                return true;

            if (key.StartsWith("DANG UY PHUONG BINH TIEN"))
                return true;

            if (Regex.IsMatch(key, @"^SO\s+\d+[\-/]"))
                return true;

            if (Regex.IsMatch(key, @"^BINH TIEN,\s*NGAY"))
                return true;

            return false;
        }

        private bool IsStopLineForTitleBlock(string line, int collectedCount)
        {
            if (string.IsNullOrWhiteSpace(line))
                return true;

            string key = FoldVietnameseKey(line);

            if (Regex.IsMatch(key, @"^(PHAN\s+THU|KET\s+QUA\s+THUC\s+HIEN)\b"))
                return true;

            if (Regex.IsMatch(key, @"^([IVX]+|A|B|C|D)\.?\s+(TINH\s+HINH|DAC\s+DIEM|KET\s+QUA|NHIEM\s+VU|TO\s+CHUC|CONG\s+TAC)\b"))
                return true;

            if (Regex.IsMatch(key, @"^\d+\.\s+(DAC\s+DIEM|CONG\s+TAC|VIEC|TINH\s+HINH|KET\s+QUA)\b"))
                return true;

            if (key.StartsWith("CAN CU ") ||
                key.StartsWith("XET ") ||
                key.StartsWith("DIEU ") ||
                key.StartsWith("KINH GUI "))
                return true;

            // Dòng tiêu đề có thể bắt đầu bằng "thực hiện Chỉ thị...".
            // Chỉ dừng "Thực hiện Công văn/Kế hoạch..." khi đã lấy được ít nhất 1 dòng tiêu đề phía trên.
            // Không dừng "Thực hiện Chỉ thị..." nếu nó là dòng đầu trích yếu dưới KẾ HOẠCH.
            // Ví dụ: KẾ HOẠCH / tiếp tục thực hiện Chỉ thị số 28-CT/TW...
            if (collectedCount > 0 &&
                Regex.IsMatch(key, @"^THUC\s+HIEN\s+(CONG\s+VAN|KE\s+HOACH|QUYET\s+DINH|NGHI\s+QUYET)\b"))
                return true;

            if (key.StartsWith("STT ") || key.Contains(" TEN CONG VIEC "))
                return true;
            // Khi đã lấy được ít nhất 1 dòng trích yếu,
            // nếu gặp dòng mở đầu thân văn bản thì dừng.
            // Không chặn dòng đầu tiên vì có nhiều trích yếu hợp lệ bắt đầu bằng:
            // "tiếp tục thực hiện...", "thực hiện..."
            if (collectedCount > 0)
            {
                if (Regex.IsMatch(key,
                    @"^THUC\s+HIEN\s+(CHI\s+THI|CONG\s+VAN|KE\s+HOACH|QUYET\s+DINH|NGHI\s+QUYET|THONG\s+BAO)\s+SO\s+\d{1,6}.*(;|NHU\s+SAU)",
                    RegexOptions.IgnoreCase))
                    return true;

                if (Regex.IsMatch(key,
                    @"^(CAN\s+CU|XET|KINH\s+GUI|NOI\s+DUNG)\b",
                    RegexOptions.IgnoreCase))
                    return true;

                if (Regex.IsMatch(key,
                    @"^(BAN\s+THUONG\s+VU|DANG\s+UY).*(XAY\s+DUNG|BAN\s+HANH|TRIEN\s+KHAI).*(NHU\s+SAU)",
                    RegexOptions.IgnoreCase))
                    return true;

                if (Regex.IsMatch(key,
                    @"^([IVX]+|\d+)\.\s+(MUC\s+DICH|YEU\s+CAU|NOI\s+DUNG|TO\s+CHUC|THUC\s+HIEN)\b",
                    RegexOptions.IgnoreCase))
                    return true;
            }

            return false;
        }

        private bool IsPossibleTitleLineForExtract(string line)
        {
            if (string.IsNullOrWhiteSpace(line))
                return false;

            string key = FoldVietnameseKey(line);

            if (line.Length < 3)
                return false;

            if (IsDocumentTypeHeaderForTitleExtract(line))
                return false;

            if (IsSkipNoiseLineForTitleExtract(line))
                return false;

            if (Regex.IsMatch(key, @"^SO\s+\d+[\-/]"))
                return false;

            if (Regex.IsMatch(key, @"^\d+$"))
                return false;

            return true;
        }

        private bool IsBetterTrichYeuTitle(string strictTitle, string currentTitle)
        {
            strictTitle = CleanFinalTrichYeuTitle(strictTitle);
            currentTitle = CleanFinalTrichYeuTitle(currentTitle);

            if (string.IsNullOrWhiteSpace(strictTitle))
                return false;

            if (string.IsNullOrWhiteSpace(currentTitle))
                return true;

            string strictKey = FoldVietnameseKey(strictTitle);
            string currentKey = FoldVietnameseKey(currentTitle);

            // Sửa lỗi lấy dư xuống phần nội dung.
            if (currentKey.Contains(" PHAN THU ") ||
                currentKey.Contains(" KET QUA THUC HIEN ") ||
                currentKey.Contains(" I. TINH HINH ") ||
                currentKey.Contains(" A. TINH HINH ") ||
                currentKey.Contains(" CAN CU ") ||
                currentKey.Contains(" THUC HIEN CONG VAN ") ||
                currentKey.Contains(" STT TEN CONG VIEC "))
                return true;

            // Sửa lỗi lấy thiếu đầu tiêu đề.
            if (strictTitle.Length > currentTitle.Length + 15 &&
                strictKey.Contains(currentKey))
                return true;

            // Nếu current quá dài nhưng strict là block tiêu đề gọn hơn thì ưu tiên strict.
            if (currentTitle.Length > strictTitle.Length + 25 &&
                strictTitle.Length >= 12)
                return true;

            return false;
        }

        private string TrimWrongBodyPartFromTitle(string value)
        {
            if (string.IsNullOrWhiteSpace(value))
                return "";

            string s = value.Trim();

            string[] patterns =
            {
                @"\s+PHẦN\s+THỨ\s+.*$",
                @"\s+PHAN\s+THU\s+.*$",
                @"\s+KẾT\s+QUẢ\s+THỰC\s+HIỆN\s+.*$",
                @"\s+KET\s+QUA\s+THUC\s+HIEN\s+.*$",
                @"\s+[IVX]+\.\s+TÌNH\s+HÌNH\s+.*$",
                @"\s+[IVX]+\.\s+TINH\s+HINH\s+.*$",
                @"\s+A\.\s+TÌNH\s+HÌNH\s+.*$",
                @"\s+A\.\s+TINH\s+HINH\s+.*$",
                @"\s+1\.\s+Đặc\s+điểm\s+.*$",
                @"\s+1\.\s+Dac\s+diem\s+.*$",
                @"\s+Căn\s+cứ\s+.*$",
                @"\s+Can\s+cu\s+.*$",
                @"\s+Thực\s+hiện\s+Công\s+văn\s+.*$",
                @"\s+Thuc\s+hien\s+Cong\s+van\s+.*$",
                @"\s+STT\s+Tên\s+công\s+việc\s+.*$",
                @"\s+STT\s+Ten\s+cong\s+viec\s+.*$"
            };

            foreach (string p in patterns)
            {
                s = Regex.Replace(s, p, "", RegexOptions.IgnoreCase);
            }

            return CleanFinalTrichYeuTitle(s);
        }

        private string CleanFinalTrichYeuTitle(string value)
        {
            if (string.IsNullOrWhiteSpace(value))
                return "";

            string s = value.Trim();

            s = Regex.Replace(s, @"\s+", " ");
            s = Regex.Replace(s, @"\s+([,.;:])", "$1");
            s = Regex.Replace(s, @"\(\s+", "(");
            s = Regex.Replace(s, @"\s+\)", ")");
            s = Regex.Replace(s, @"[\-\–\—\s]+$", "");

            return s.Trim();
        }

        private string FoldVietnameseKey(string input)
        {
            if (string.IsNullOrWhiteSpace(input))
                return "";

            string normalized = input.Normalize(System.Text.NormalizationForm.FormD);
            var sb = new System.Text.StringBuilder();

            foreach (char c in normalized)
            {
                var category = System.Globalization.CharUnicodeInfo.GetUnicodeCategory(c);

                if (category != System.Globalization.UnicodeCategory.NonSpacingMark)
                    sb.Append(c);
            }

            string s = sb.ToString().Normalize(System.Text.NormalizationForm.FormC);

            s = s.Replace('Đ', 'D').Replace('đ', 'd');
            s = s.ToUpperInvariant();
            s = Regex.Replace(s, @"\s+", " ");

            return s.Trim();
        }

        private string NormalizeTrichYeuTitleFromBadOcr(string trichYeu, string rawOcrText)
        {
            string source = "";

            if (!string.IsNullOrWhiteSpace(trichYeu))
                source += " " + trichYeu;

            if (!string.IsNullOrWhiteSpace(rawOcrText))
                source += " " + rawOcrText;

            if (string.IsNullOrWhiteSpace(source))
                return trichYeu ?? "";

            string norm = NormalizeVietnameseLoose(source);

            /*
             * FIX RIÊNG CHO MẪU TRONG ẢNH:
             * OCR sai:
             * CÔNG TÁC NDC CUÍNH, PHÙNG THÔNH TII AN NHŨNG,
             * LÀNG PHÍ TIÊU CHỨC VÀ CẢI CÁCHTN PHÁP
             *
             * PDF đúng:
             * VỀ CÔNG TÁC NỘI CHÍNH, PHÒNG CHỐNG THAM NHŨNG,
             * LÃNG PHÍ TIÊU CỰC VÀ CẢI CÁCH TƯ PHÁP
             */
            // Chỉ ép sửa cứng khi OCR có dấu hiệu lỗi chữ thật sự.
            // Không ép các trích yếu đã OCR đúng như:
            // "Về công tác nội chính, phòng, chống tham nhũng, tiêu cực và cải cách tư pháp tháng 11 năm 2025".
            bool hasBadNoiChinhOcr =
                norm.Contains("ndc cuinh") ||
                norm.Contains("ndc chinh") ||
                norm.Contains("nui chinh") ||
                norm.Contains("phung thonh") ||
                norm.Contains("phong thonh") ||
                norm.Contains("tii an nhung") ||
                norm.Contains("an nhung") ||
                norm.Contains("lang phi tieu chuc") ||
                norm.Contains("cai cachtn") ||
                norm.Contains("tn phap");

            bool looksLikeNoiChinhTitle =
                hasBadNoiChinhOcr &&
                norm.Contains("cong tac") &&
                (
                    norm.Contains("noi chinh") ||
                    norm.Contains("ndc cuinh") ||
                    norm.Contains("ndc chinh") ||
                    norm.Contains("nui chinh")
                ) &&
                (
                    norm.Contains("tham nhung") ||
                    norm.Contains("an nhung")
                ) &&
                (
                    norm.Contains("lang phi") ||
                    norm.Contains("lang phi tieu chuc") ||
                    norm.Contains("lang phi tieu cuc")
                ) &&
                (
                    norm.Contains("cai cach") ||
                    norm.Contains("cai cachtn")
                ) &&
                (
                    norm.Contains("tu phap") ||
                    norm.Contains("tn phap")
                );

            if (looksLikeNoiChinhTitle)
            {
                return "VỀ CÔNG TÁC NỘI CHÍNH, PHÒNG CHỐNG THAM NHŨNG, LÃNG PHÍ TIÊU CỰC VÀ CẢI CÁCH TƯ PHÁP";
            }

            /*
             * Sửa nhẹ các lỗi OCR phổ biến, không ép cứng nếu không nhận ra mẫu.
             */
            string result = trichYeu ?? "";

            result = Regex.Replace(result, @"\bNDC\s+CUÍNH\b", "NỘI CHÍNH", RegexOptions.IgnoreCase);
            result = Regex.Replace(result, @"\bNDC\s+CHÍNH\b", "NỘI CHÍNH", RegexOptions.IgnoreCase);
            result = Regex.Replace(result, @"\bNUI\s+CHÍNH\b", "NỘI CHÍNH", RegexOptions.IgnoreCase);

            result = Regex.Replace(result, @"\bPHÙNG\s+THÔNH\b", "PHÒNG CHỐNG", RegexOptions.IgnoreCase);
            result = Regex.Replace(result, @"\bPHÒNG\s+THÔNH\b", "PHÒNG CHỐNG", RegexOptions.IgnoreCase);

            result = Regex.Replace(result, @"\bTII\s+AN\s+NHŨNG\b", "THAM NHŨNG", RegexOptions.IgnoreCase);
            result = Regex.Replace(result, @"\bAN\s+NHŨNG\b", "THAM NHŨNG", RegexOptions.IgnoreCase);

            result = Regex.Replace(result, @"\bLÀNG\s+PHÍ\b", "LÃNG PHÍ", RegexOptions.IgnoreCase);
            result = Regex.Replace(result, @"\bTIÊU\s+CHỨC\b", "TIÊU CỰC", RegexOptions.IgnoreCase);

            result = Regex.Replace(result, @"\bCẢI\s+CÁCHTN\s+PHÁP\b", "CẢI CÁCH TƯ PHÁP", RegexOptions.IgnoreCase);
            result = Regex.Replace(result, @"\bCẢI\s+CÁCH\s*TN\s+PHÁP\b", "CẢI CÁCH TƯ PHÁP", RegexOptions.IgnoreCase);

            result = Regex.Replace(result, @"\s+", " ").Trim();

            return result;
        }

        private string ExtractTrichYeuNoiDungFromText(string rawText)
        {
            if (string.IsNullOrWhiteSpace(rawText))
                return "";

            // Bước 1: Thử tìm trực tiếp bằng regex — nhanh và chính xác hơn.
            // Xử lý được cả trích yếu 1 dòng lẫn nhiều dòng.
            string directResult = ExtractTrichYeuByDirectSearch(rawText);
            if (!string.IsNullOrWhiteSpace(directResult))
                return directResult;

            // BỔ SUNG: Trường hợp Công văn / văn bản hành chính không có dòng tiêu đề lớn.
            // Trích yếu nằm ngay dưới dòng Số/Ký hiệu, ví dụ:
            // Số 237-CV/ĐU
            // Về việc xác minh thái độ chính trị hiện nay
            // phục vụ công tác cán bộ
            string underNumberResult = ExtractTrichYeuUnderSoKyHieu(rawText);
            if (!string.IsNullOrWhiteSpace(underNumberResult))
                return underNumberResult;

            // Bước 2: Fallback — tìm theo từng dòng (giữ nguyên logic cũ).

            // --- Fallback line-by-line ---
            string normalized = rawText
                .Replace("\r", "\n")
                .Replace("\u00A0", " ");

            normalized = Regex.Replace(normalized, @"[ \t]+", " ");

            var lines = normalized
                .Split(new[] { '\n' }, StringSplitOptions.RemoveEmptyEntries)
                .Select(x => CleanTrichYeuLine(x))
                .Where(x => !string.IsNullOrWhiteSpace(x))
                .Where(x => !IsUiNoiseLine(x))
                .ToList();

            if (lines.Count == 0)
                return "";

            int titleIndex = -1;

            // Chỉ lấy tiêu đề văn bản thật trong OCR: QUYẾT ĐỊNH, BÁO CÁO, KẾ HOẠCH...
            // Tuyệt đối không lấy loại văn bản trong form như Chỉ thị.
            for (int i = 0; i < lines.Count; i++)
            {
                string line = lines[i];

                if (!IsDocumentTitleLine(line))
                    continue;

                // Bỏ các dòng menu/UI.
                if (IsUiNoiseLine(line))
                    continue;

                // Ưu tiên dòng tiêu đề nằm sau dòng số văn bản.
                bool hasSoVanBanBefore = false;
                int from = Math.Max(0, i - 10);

                for (int k = from; k < i; k++)
                {
                    string prev = lines[k];

                    if (Regex.IsMatch(prev,
                        @"^(Số|So|SỐ|SO)?\s*\d{1,6}\s*[-–—]\s*[A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]+(?:\/[A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]+)+",
                        RegexOptions.IgnoreCase))
                    {
                        hasSoVanBanBefore = true;
                        break;
                    }
                }

                if (hasSoVanBanBefore)
                {
                    titleIndex = i;
                    break;
                }

                if (titleIndex < 0)
                    titleIndex = i;
            }

            if (titleIndex < 0)
                return "";

            var parts = new System.Collections.Generic.List<string>();

            string titleLine = lines[titleIndex];

            // Trường hợp OCR dính chung: "QUYẾT ĐỊNH về việc..."
            string sameLineAfterTitle = Regex.Replace(
                titleLine,
                @"^\s*(QUYẾT\s*ĐỊNH|QUYET\s*DINH|BÁO\s*CÁO|BAO\s*CAO|KẾ\s*HOẠCH|KE\s*HOACH|THÔNG\s*BÁO|THONG\s*BAO|TỜ\s*TRÌNH|TO\s*TRINH|NGHỊ\s*QUYẾT|NGHI\s*QUYET|CÔNG\s*VĂN|CONG\s*VAN|CHƯƠNG\s*TRÌNH|CHUONG\s*TRINH|QUY\s*CHẾ|QUY\s*CHE|HƯỚNG\s*DẪN|HUONG\s*DAN|BIÊN\s*BẢN|BIEN\s*BAN|CHỈ\s*THỊ|CHI\s*THI)\b",
                "",
                RegexOptions.IgnoreCase
            );

            sameLineAfterTitle = CleanTrichYeuLine(sameLineAfterTitle);

            if (IsValidTrichYeuCandidate(sameLineAfterTitle))
                parts.Add(sameLineAfterTitle);

            // Lấy đúng các dòng ngay dưới tiêu đề, nối đủ cả cụm được bôi đen.
            for (int j = titleIndex + 1; j < lines.Count && j <= titleIndex + 12; j++)
            {
                string candidate = CleanTrichYeuLine(lines[j]);

                if (string.IsNullOrWhiteSpace(candidate))
                {
                    if (parts.Count == 0) continue;
                    break;
                }

                // Dừng trước phần căn cứ/nội dung quyết định.
                if (IsTrichYeuStopLine(lines[j], candidate))
                    break;

                // Nếu gặp lại tiêu đề văn bản khác thì dừng.
                if (IsDocumentTitleLine(candidate))
                {
                    if (parts.Count == 0) continue;
                    break;
                }

                // Sau dòng tiêu đề chỉ dừng khi gặp UI noise thật sự.
                // Không dừng vì "đảng ủy phường" hay các từ nội dung — cụm đó có thể nằm trong trích yếu.
                if (IsUiNoiseLine(candidate))
                {
                    if (parts.Count == 0) continue;
                    break;
                }

                // Dòng đầu tiên sau tiêu đề không được là tên loại văn bản đơn thuần (không có nội dung).
                if (parts.Count == 0 && IsStandaloneDocTypeLine(candidate))
                    continue;

                parts.Add(candidate);

                // Nếu dòng kế tiếp là Căn cứ/Xét/Điều thì dừng sau khi đã lấy đủ dòng hiện tại.
                if (j + 1 < lines.Count)
                {
                    string next = CleanTrichYeuLine(lines[j + 1]);

                    if (IsTrichYeuStopLine(lines[j + 1], next))
                        break;
                }
            }

            string result = CleanTrichYeuLine(string.Join(" ", parts));

            // Dọn lỗi OCR khoảng trắng.
            result = Regex.Replace(result, @"\s+", " ").Trim();

            // Kiểm tra nhẹ: không dùng IsValidTrichYeuCandidate vì nó sẽ chặn nhầm
            // trích yếu hợp lệ có chứa "Đảng ủy phường Bình Tiên".
            if (string.IsNullOrWhiteSpace(result) || result.Length < 6)
                return "";

            if (IsUiNoiseLine(result))
                return "";

            return result;
        }

        /// <summary>
        /// Bổ sung cho trường hợp trích yếu nằm ngay dưới dòng Số/Ký hiệu văn bản.
        /// Ví dụ:
        /// Số 237-CV/ĐU
        /// Về việc xác minh thái độ chính trị hiện nay
        /// phục vụ công tác cán bộ
        /// </summary>
        private string ExtractTrichYeuUnderSoKyHieu(string rawText)
        {
            if (string.IsNullOrWhiteSpace(rawText))
                return "";

            string normalized = rawText
                .Replace("\r", "\n")
                .Replace("\u00A0", " ");

            normalized = Regex.Replace(normalized, @"[ \t]+", " ");

            var lines = normalized
                .Split(new[] { '\n' }, StringSplitOptions.RemoveEmptyEntries)
                .Select(x => CleanTrichYeuLine(x))
                .Where(x => !string.IsNullOrWhiteSpace(x))
                .Where(x => !IsUiNoiseLine(x))
                .ToList();

            if (lines.Count == 0)
                return "";

            int soIndex = -1;
            int maxHeaderScan = Math.Min(lines.Count, 25);

            // Ưu tiên bắt dòng Số/Ký hiệu ở phần đầu văn bản, không bắt "Công văn số..." trong thân.
            for (int i = 0; i < maxHeaderScan; i++)
            {
                string line = lines[i];

                if (Regex.IsMatch(line,
                    @"^S[oố]\s*[:：]?\s*\d{1,6}\s*[-–—]\s*[A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]{1,10}(?:\s*[\/\-]\s*[A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]{1,15})+",
                    RegexOptions.IgnoreCase))
                {
                    soIndex = i;
                    break;
                }
            }

            if (soIndex < 0)
                return "";

            var parts = new System.Collections.Generic.List<string>();

            for (int j = soIndex + 1; j < lines.Count && j <= soIndex + 10; j++)
            {
                string candidate = CleanTrichYeuLine(lines[j]);

                if (string.IsNullOrWhiteSpace(candidate))
                {
                    if (parts.Count == 0) continue;
                    break;
                }

                // Bỏ qua dòng KHẨN / HỎA TỐC nằm trong header
                if (Regex.IsMatch(FoldVietnameseKey(candidate),
                    @"^(KHAN|THUONG KHAN|HOA TOC)$"))
                {
                    continue;
                }

                string lower = Regex.Replace(candidate.ToLowerInvariant(), @"\s+", " ").Trim();

                // Dừng trước mục đánh số, "Nay,", "Nội dung như sau"
                // FIX: OCR có thể tách "(nhiệm kỳ 2025 - 2030)" thành:
                // "(nhiệm kỳ 2025" và dòng sau "- 2030)"
                // CleanTrichYeuLine sẽ làm "- 2030)" thành "2030)", không được hiểu là mục đánh số.
                if (Regex.IsMatch(candidate, @"^\d{4}\s*\)$", RegexOptions.IgnoreCase) &&
                    parts.Count > 0 &&
                    Regex.IsMatch(parts[parts.Count - 1], @"\(\s*nhiệm\s*kỳ\s+\d{4}$", RegexOptions.IgnoreCase))
                {
                    parts[parts.Count - 1] = parts[parts.Count - 1] + " - " + candidate;
                    break;
                }

                // Dừng trước mục đánh số, nhưng không dừng nhầm năm kết thúc nhiệm kỳ.
                if (Regex.IsMatch(candidate, @"^\d+\s*[\.\)]", RegexOptions.IgnoreCase))
                    break;
                if (Regex.IsMatch(FoldVietnameseKey(candidate),
                    @"^(NAY[,\s]|NOI DUNG NHU SAU|T/M|TM DANG UY)"))
                    break;

                /*
 * FIX: Dòng "(Ngày ... tháng ... năm ...)" ngay dưới trích yếu
 * là một phần của trích yếu, không được bỏ.
 *
 * Nhưng dòng địa danh/ngày tháng ở header như:
 * "Bình Tiên, ngày 18 tháng 11 năm 2025"
 * thì vẫn phải bỏ.
 */
                bool isParenthesizedDateSubtitle = Regex.IsMatch(
                    lower,
                    @"^\(?\s*(ngày|ngay)\s+\d{1,2}\s+(tháng|thang)\s+\d{1,2}\s+(năm|nam)\s+\d{4}\s*\)?$",
                    RegexOptions.IgnoreCase
                );

                if (isParenthesizedDateSubtitle)
                {
                    if (parts.Count > 0)
                    {
                        parts.Add(candidate);
                        break;
                    }

                    continue;
                }

                // Bỏ dòng ngày tháng nằm cạnh phần đầu văn bản/header.
                if (lower.StartsWith("bình tiên,") || lower.StartsWith("binh tien,"))
                {
                    if (parts.Count == 0) continue;
                    break;
                }

                // Nếu đã bắt đầu lấy trích yếu rồi thì dòng có ngày/tháng/năm vẫn có thể là phần tiếp nối tiêu đề.
                // Ví dụ: "Kết luận số 183-KL/TW ngày 01 tháng 8 năm 2025"
                if ((lower.Contains("ngày") && lower.Contains("tháng") && lower.Contains("năm")) ||
                    (lower.Contains("ngay") && lower.Contains("thang") && lower.Contains("nam")))
                {
                    if (parts.Count == 0) continue;

                    parts.Add(candidate);
                    continue;
                }

                // Bỏ các dòng tiêu đề loại văn bản nếu có.
                if (IsStandaloneDocTypeLine(candidate) || IsDocumentTitleLine(candidate))
                {
                    if (parts.Count == 0) continue;
                    break;
                }

                // Dừng trước phần kính gửi hoặc phần thân văn bản.
                if (lower.StartsWith("kính gửi") ||
                    lower.StartsWith("kinh gui") ||
                    lower.StartsWith("căn cứ") ||
                    lower.StartsWith("can cu") ||
                    lower.StartsWith("để ") ||
                    lower.StartsWith("de ") ||
                    lower.StartsWith("xét ") ||
                    lower.StartsWith("xet ") ||
                    lower.StartsWith("đồng chí") ||
                    lower.StartsWith("dong chi") ||
                    lower.StartsWith("rất mong") ||
                    lower.StartsWith("rat mong"))
                {
                    break;
                }

                if (IsTrichYeuStopLine(lines[j], candidate))
                    break;

                if (!IsValidTrichYeuCandidate(candidate))
                {
                    /*
                     * FIX: Dòng tiếp nối của trích yếu vẫn có thể chứa
                     * 'Đảng bộ Thành phố', ví dụ:
                     * 'nộp lưu trữ lịch sử Đảng bộ Thành phố'
                     *
                     * IsValidTrichYeuCandidate() đang chặn cụm này để tránh lấy nhầm header,
                     * nhưng khi đã bắt đầu lấy trích yếu rồi thì phải cho phép dòng tiếp nối hợp lệ.
                     */
                    if (parts.Count > 0 && IsValidTrichYeuContinuationLine(candidate))
                    {
                        parts.Add(candidate);
                    }
                    else
                    {
                        if (parts.Count == 0) continue;
                        break;
                    }
                }
                else
                {
                    parts.Add(candidate);
                }

                // Nếu đã lấy được dòng bắt đầu bằng "Về việc..." thì cho phép nối thêm dòng kế tiếp,
                // nhưng dừng khi dòng sau chuyển sang Kính gửi / nội dung thân văn bản.
                if (j + 1 < lines.Count)
                {
                    string next = CleanTrichYeuLine(lines[j + 1]);
                    string nextLower = Regex.Replace(next.ToLowerInvariant(), @"\s+", " ").Trim();

                    if (nextLower.StartsWith("kính gửi") ||
                        nextLower.StartsWith("kinh gui") ||
                        nextLower.StartsWith("căn cứ") ||
                        nextLower.StartsWith("can cu") ||
                        nextLower.StartsWith("để ") ||
                        nextLower.StartsWith("de ") ||
                        nextLower.StartsWith("xét ") ||
                        nextLower.StartsWith("xet ") ||
                        nextLower.StartsWith("đồng chí") ||
                        nextLower.StartsWith("dong chi") ||
                        nextLower.StartsWith("rất mong") ||
                        nextLower.StartsWith("rat mong"))
                    {
                        break;
                    }
                }
            }

            string result = CleanTrichYeuLine(string.Join(" ", parts));
            result = Regex.Replace(result, @"\s+", " ").Trim();

            if (string.IsNullOrWhiteSpace(result) || result.Length < 6)
                return "";

            if (IsUiNoiseLine(result))
                return "";

            if (IsStandaloneDocTypeLine(result))
                return "";

            return result;
        }

        private bool IsDocumentTitleLine(string line)
        {
            if (string.IsNullOrWhiteSpace(line)) return false;

            string s = Regex.Replace(line.Trim(), @"\s+", " ");
            string lower = s.ToLowerInvariant();

            // FIX: chặn menu UI bị nhận nhầm là tiêu đề văn bản
            if (IsUiNoiseLine(s))
                return false;

            string upper = s.ToUpperInvariant();
            return Regex.IsMatch(upper,
                            @"^(QUYẾT\s*ĐỊNH|QUYẾT\s*ĐINH|QUYET\s*DINH|BÁO\s*CÁO|BAO\s*CAO|KẾ\s*HOẠCH|KE\s*HOACH|THÔNG\s*BÁO|THONG\s*BAO|TỜ\s*TRÌNH|TO\s*TRINH|NGHỊ\s*QUYẾT|NGHI\s*QUYET|CÔNG\s*VĂN|CONG\s*VAN|CHƯƠNG\s*TRÌNH|CHUONG\s*TRINH|QUY\s*CHẾ|QUY\s*CHE|HƯỚNG\s*DẪN|HUONG\s*DAN|BIÊN\s*BẢN|BIEN\s*BAN|CHỈ\s*THỊ|CHI\s*THI)\b");
        }

        /// <summary>
        /// Trả về true nếu dòng chỉ là tên loại văn bản đơn lẻ, không có nội dung trích yếu.
        /// Dùng để bỏ qua dòng đó khi nó xuất hiện ngay sau dòng tiêu đề.
        /// Ví dụ: dòng chỉ là "Quyết định" hoặc "Chỉ thị" thì bỏ qua.
        /// </summary>
        private bool IsStandaloneDocTypeLine(string s)
        {
            if (string.IsNullOrWhiteSpace(s)) return false;

            string lower = CleanTrichYeuLine(s).ToLowerInvariant().Trim();

            string[] titleOnly =
            {
                "quyết định", "quyet dinh",
                "báo cáo",    "bao cao",
                "kế hoạch",   "ke hoach",
                "thông báo",  "thong bao",
                "tờ trình",   "to trinh",
                "nghị quyết", "nghi quyet",
                "công văn",   "cong van",
                "chương trình","chuong trinh",
                "quy chế",    "quy che",
                "hướng dẫn",  "huong dan",
                "biên bản",   "bien ban",
                "chỉ thị",    "chi thi"
            };

            return Array.Exists(titleOnly, t => lower == t);
        }

        private string CleanTrichYeuLine(string line)
        {
            if (line == null) return "";

            string s = Regex.Replace(line, @"\s+", " ").Trim();

            // Gặp dấu chấm/gạch kéo dài thì chỉ lấy phần trước đó.
            // Ví dụ: "Thành lập Đảng bộ ... -----" => "Thành lập Đảng bộ"
            s = Regex.Split(s, @"(\.{3,}|…+|[-–—_]{3,}|={3,})")[0];

            s = s.Trim('-', '–', '—', '_', '=', '*', '.', ':', ';', ' ');
            return s.Trim();
        }

        private bool IsTrichYeuStopLine(string line)
        {
            return IsTrichYeuStopLine(line, CleanTrichYeuLine(line));
        }

        private string RemoveVietnameseTone(string input)
        {
            return FoldVietnameseKey(input).ToLowerInvariant();
        }

        private bool IsTrichYeuStopLine(string originalLine, string cleanedLine)
        {
            string original = originalLine == null ? "" : originalLine.Trim();
            string s = cleanedLine == null ? "" : cleanedLine.Trim();
            string lower = s.ToLowerInvariant();
            // SỬA/THÊM: ràng buộc dừng chung để không lấy lố sang phần thân văn bản
            string x = RemoveVietnameseTone(s ?? "").ToLowerInvariant();
            x = Regex.Replace(x, @"\s+", " ").Trim();

            if (string.IsNullOrWhiteSpace(x))
                return true;

            // Dừng trước các dòng mở đầu phần thân văn bản
            if (Regex.IsMatch(x, @"^(kinh gui|can cu|xet|theo|thuc hien|ngay|noi nhan|tran trong|tm\.|t/m|nguoi ky|bi thu|pho bi thu)\b"))
                return true;

            // Dừng trước các dòng kiểu: "Nội dung ... như sau", "... gồm:"
            if (Regex.IsMatch(x, @"\b(noi dung|noi dung xac minh|noi dung bao cao|noi dung trien khai)\b.*\b(nhu sau|gom)\b"))
                return true;

            if (Regex.IsMatch(x, @"\b(nhu sau|gom:|gom)\s*$"))
                return true;

            // Dừng trước các mục đánh số thân văn bản: 1. Bản thân, 2. Cha ruột, I., II., Điều 1...
            if (Regex.IsMatch(x, @"^(\d+[\.\)]|[ivx]+[\.\)]|dieu\s+\d+)\s+"))
                return true;

            if (Regex.IsMatch(original, @"^[-–—_=*. ]+$")) return true;
            if (Regex.IsMatch(original, @"(\.{3,}|…+|[-–—_]{3,}|={3,})") && string.IsNullOrWhiteSpace(s)) return true;

            if (lower.StartsWith("căn cứ") || lower.StartsWith("can cu")) return true;
            if (lower.StartsWith("xét ") || lower.StartsWith("xet ")) return true;
            if (lower.StartsWith("ban thường vụ") || lower.StartsWith("ban thuong vu")) return true;
            if (lower.StartsWith("ban chấp hành") || lower.StartsWith("ban chap hanh")) return true;
            if (lower.StartsWith("điều ") || lower.StartsWith("dieu ")) return true;
            // FIX: Dừng trước các mục nội dung như "1. Công tác...", "2. Việc..."
            if (Regex.IsMatch(s, @"^\d+\s*[\.\)]\s+", RegexOptions.IgnoreCase)) return true;
            if (Regex.IsMatch(s, @"^[IVXLCDM]+\s*[\.\)]\s+", RegexOptions.IgnoreCase)) return true;
            return false;
        }

        private bool IsValidTrichYeuContinuationLine(string s)
        {
            // SỬA/THÊM: nếu dòng tiếp theo đã là phần thân thì không cho nối vào trích yếu
            if (IsTrichYeuStopLine(s))
                return false;

            if (string.IsNullOrWhiteSpace(s))
                return false;

            s = CleanTrichYeuLine(s);

            if (string.IsNullOrWhiteSpace(s))
                return false;

            if (s.Length < 4)
                return false;

            if (IsUiNoiseLine(s))
                return false;

            if (IsStandaloneDocTypeLine(s))
                return false;

            if (Regex.IsMatch(s, @"^[-–—_=*. ]+$"))
                return false;

            string lower = Regex.Replace(s.ToLowerInvariant(), @"\s+", " ").Trim();

            // Vẫn phải chặn các dòng chuyển sang phần thân văn bản.
            if (lower.StartsWith("kính gửi")) return false;
            if (lower.StartsWith("kinh gui")) return false;
            if (lower.StartsWith("căn cứ")) return false;
            if (lower.StartsWith("can cu")) return false;
            if (lower.StartsWith("xét ")) return false;
            if (lower.StartsWith("xet ")) return false;
            if (lower.StartsWith("đồng chí")) return false;
            if (lower.StartsWith("dong chi")) return false;
            if (lower.StartsWith("rất mong")) return false;
            if (lower.StartsWith("rat mong")) return false;
            if (lower.StartsWith("thực hiện")) return false;
            if (lower.StartsWith("thuc hien")) return false;

            // Chặn header thật, nhưng KHÔNG chặn dòng nội dung có 'Đảng bộ Thành phố'.
            if (lower.Contains("đảng cộng sản việt nam")) return false;
            if (lower.Contains("dang cong san viet nam")) return false;

            if (lower.StartsWith("đảng bộ thành phố")) return false;
            if (lower.StartsWith("dang bo thanh pho")) return false;

            if (lower.StartsWith("đảng ủy phường")) return false;
            if (lower.StartsWith("dang uy phuong")) return false;

            if (lower.StartsWith("bình tiên,")) return false;
            if (lower.StartsWith("binh tien,")) return false;

            if (lower.StartsWith("số ")) return false;
            if (lower.StartsWith("so ")) return false;

            return true;
        }


        private bool IsValidTrichYeuCandidate(string s)
        {
            if (string.IsNullOrWhiteSpace(s))
                return false;

            s = CleanTrichYeuLine(s);

            if (string.IsNullOrWhiteSpace(s))
                return false;

            if (IsUiNoiseLine(s))
                return false;

            if (s.Length < 6)
                return false;

            if (Regex.IsMatch(s, @"^[-–—_=*. ]+$"))
                return false;

            string lower = Regex.Replace(s.ToLowerInvariant(), @"\s+", " ").Trim();

            // Chặn chắc chắn các dòng UI/form/menu.
            string[] bad =
            {
        "danh mục riêng",
        "danh mục chung",
        "kho lưu trữ",
        "báo cáo thống kê",
        "bao cao thong ke",
        "số hóa tài liệu",
        "số hóa hồ sơ",
        "quản lý biên mục",
        "duyệt biên mục",
        "biên mục bổ sung",
        "phê duyệt hồ sơ",
        "duyệt lưu kho",
        "thông tin tài liệu",
        "khung biên mục",
        "mã định danh tài liệu",
        "mã hồ sơ",
        "tiêu đề/tên hồ sơ",
        "số thứ tự văn bản",
        "trích yếu nội dung",
        "tác giả văn bản",
        "loại bản",

        private string TryGetFullTextContent()
        {
            try
            {
                /*
                 * Sửa đúng lỗi không đọc được:
                 * - Không chỉ bấm Nội dung toàn văn một lần rồi đọc ngay.
                 * - Phải bấm Nhận dạng OCR, chờ OCR thật sự chạy xong, sau đó thử mở Nội dung toàn văn nhiều lần.
                 * - Chỉ lấy text trong popup/toàn văn hoặc vùng bên phải, tránh lấy nhầm form bên trái.
                 */
                if (!TryRunNhanDangOcr())
                    return "";

                DateTime timeout = DateTime.Now.AddSeconds(75);
                string bestText = "";

                while (DateTime.Now < timeout)
                {
                    AddLog("OCR", "Đang làm", "Bấm Nội dung toàn văn để lấy text OCR.");

                    bool clicked = false;

                    try
                    {
                        object jsClicked = ((IJavaScriptExecutor)driver).ExecuteScript(@"
                            function visible(el) {
                                if (!el) return false;
                                const r = el.getBoundingClientRect();
                                const st = window.getComputedStyle(el);
                                return r.width > 0 && r.height > 0 &&
                                       st.display !== 'none' &&
                                       st.visibility !== 'hidden';
                            }

                            let items = Array.from(document.querySelectorAll('button, a, span, div, li'))
                                .filter(x => visible(x) && (x.innerText || '').replace(/\s+/g,' ').trim().includes('Nội dung toàn văn'))
                                .map(x => ({ el: x, r: x.getBoundingClientRect() }))
                                .filter(x => x.r.left > window.innerWidth * 0.45)
                                .sort((a,b) => (a.r.width * a.r.height) - (b.r.width * b.r.height));

                            if (items.length === 0) return false;

                            let el = items[0].el;
                            el.scrollIntoView({ block: 'center', inline: 'center' });
                            el.click();
                            return true;
                        ");

                        clicked = jsClicked != null && jsClicked.ToString().ToLower() == "true";
                    }
                    catch
                    {
                        clicked = false;
                    }

                    if (!clicked)
                    {
                        AddLog("OCR", "Cảnh báo", "Không thấy nút Nội dung toàn văn ở vùng file bên phải.");
                        return "";
                    }

                    Thread.Sleep(2500);
                    try
                    {
                        wait.Until(d =>
                        {
                            try
                            {
                                object opened = ((IJavaScriptExecutor)d).ExecuteScript(@"
                function visible(el) {
                    if (!el) return false;
                    const r = el.getBoundingClientRect();
                    const st = window.getComputedStyle(el);
                    return r.width > 0 &&
                           r.height > 0 &&
                           st.display !== 'none' &&
                           st.visibility !== 'hidden';
                }

                let dialogs = Array.from(document.querySelectorAll(
                    '.modal.show .modal-content, .modal.in .modal-content, .modal-content, [role=dialog], .ui-dialog'
                ))
                .filter(visible)
                .filter(x => (x.innerText || x.textContent || '').toLowerCase().includes('nội dung toàn văn'));

                return dialogs.length > 0;
            ");

                                return opened != null && opened.ToString().ToLower() == "true";
                            }
                            catch
                            {
                                return false;
                            }
                        });
                    }
                    catch
                    {
                        AddLog("OCR", "Cảnh báo", "Đã bấm Nội dung toàn văn nhưng chưa thấy popup mở.");
                    }
                    string text = ReadNoiDungToanVanPopupText();

                    CloseNoiDungToanVanPopup();

                    Thread.Sleep(500);

                    if (!string.IsNullOrWhiteSpace(text) && text.Length > bestText.Length)
                        bestText = text;

                    if (LooksLikeDocumentText(text) || LooksLikeTextCanExtractTrichYeu(text))
                    {
                        AddLog("OCR", "OK", "Đã lấy được text OCR đủ để trích yếu.");
                        return text;
                    }

                    AddLog("OCR", "Cảnh báo", "Nội dung toàn văn chưa có text OCR hợp lệ, chờ OCR thêm rồi thử lại.");
                    Thread.Sleep(3000);
                }
                if (LooksLikeDocumentText(bestText) || LooksLikeTextCanExtractTrichYeu(bestText))
                {
                    AddLog("OCR", "OK", "Đã lấy được text OCR đủ để trích yếu.");
                    return bestText;
                }

                AddLog("OCR", "Cảnh báo", "Không lấy được text OCR hợp lệ sau khi chờ.");
                return "";
            }
            catch (Exception ex)
            {
                AddLog("OCR", "Cảnh báo", ex.Message);
                return "";
            }
        }

        private string ReadNoiDungToanVanPopupText()
        {
            try
            {
                object value = ((IJavaScriptExecutor)driver).ExecuteScript(@"
            function visible(el) {
                if (!el) return false;
                const r = el.getBoundingClientRect();
                const st = window.getComputedStyle(el);
                return r.width > 0 &&
                       r.height > 0 &&
                       st.display !== 'none' &&
                       st.visibility !== 'hidden' &&
                       st.opacity !== '0';
            }

            function clean(s) {
                return (s || '')
                    .replace(/\r/g, '\n')
                    .replace(/[ \t]+/g, ' ')
                    .replace(/\n\s*\n+/g, '\n')
                    .trim();
            }

            /*
             * CHỈ đọc modal/popup đang mở có tiêu đề Nội dung toàn văn.
             * Không đọc body, không đọc form trái, không đọc viewer.
             */
            let boxes = Array.from(document.querySelectorAll(
                '.modal.show .modal-content, ' +
                '.modal.in .modal-content, ' +
                '.modal-content, ' +
                '[role=dialog], ' +
                '.ui-dialog'
            ))
            .filter(visible)
            .filter(box => {
                let t = clean(box.innerText || box.textContent || '');
                return t.toLowerCase().includes('nội dung toàn văn');
            });

            if (boxes.length === 0) return '';

            boxes.sort((a, b) => {
                let za = parseInt(window.getComputedStyle(a).zIndex || '0');
                let zb = parseInt(window.getComputedStyle(b).zIndex || '0');
                return zb - za;
            });

            let box = boxes[0];

            /*
             * Ưu tiên textarea vì popup Nội dung toàn văn của web đang chứa OCR trong textarea.
             */
            let textareas = Array.from(box.querySelectorAll('textarea'))
                .filter(visible);

            for (let ta of textareas) {
                let t = clean(ta.value || ta.innerText || ta.textContent || '');
                t = t.replace(/^Nội dung toàn văn\s*/i, '').trim();
                t = t.replace(/\s*Đóng\s*$/i, '').trim();

                if (t.length >= 20) return t;
            }

            /*
             * Fallback: đọc mọi vùng text bên trong modal, nhưng vẫn chỉ trong modal.
             */
            let candidates = Array.from(box.querySelectorAll(
                'pre, [contenteditable=true], .modal-body, div, p, span'
            ))
            .filter(visible);

            let best = '';

            for (let el of candidates) {
                let t = clean(el.value || el.innerText || el.textContent || '');
                t = t.replace(/^Nội dung toàn văn\s*/i, '').trim();
                t = t.replace(/\s*Đóng\s*$/i, '').trim();

                if (t.length > best.length)
                    best = t;
            }

            return best;
        ");

                string text = value == null ? "" : value.ToString();

                text = text.Replace("\r", "\n");
                text = Regex.Replace(text, @"[ \t]+", " ");
                text = Regex.Replace(text, @"\n\s*\n+", "\n");
                text = text.Trim();

                return text;
            }
            catch
            {
                return "";
            }
        }

        private bool WaitOcrPdfToanVanFinished(int timeoutSeconds)
        {
            DateTime timeout = DateTime.Now.AddSeconds(timeoutSeconds);
            bool sawLoading = false;

            while (DateTime.Now < timeout)
            {
                Application.DoEvents();

                string body = "";
                try
                {
                    body = driver.FindElement(By.TagName("body")).Text.ToLowerInvariant();
                }
                catch
                {
                    body = "";
                }

                bool loading = body.Contains("đang nhận dạng") ||
                               body.Contains("dang nhan dang") ||
                               body.Contains("vui lòng chờ") ||
                               body.Contains("vui long cho") ||
                               body.Contains("đang xử lý") ||
                               body.Contains("dang xu ly") ||
                               body.Contains("loading");

                if (loading)
                    sawLoading = true;

                bool success = (body.Contains("ocr toàn văn thành công") ||
                                body.Contains("ocr toan van thanh cong") ||
                                body.Contains("ocr pdf toàn văn") && body.Contains("thành công") ||
                                body.Contains("ocr pdf toan van") && body.Contains("thanh cong"));

                // Nếu có thông báo thành công thì còn chờ overlay loading biến mất hẳn.
                if (success && !loading)
                    return true;

                Thread.Sleep(1000);
            }

            return false;
        }

        private bool LooksLikeDocumentText(string text)
        {
            if (string.IsNullOrWhiteSpace(text)) return false;

            string s = text.ToLowerInvariant();
            s = Regex.Replace(s, @"\s+", " ").Trim();

            // Loại rõ text giao diện/form trái.
            if (s.Contains("automation biên mục tài liệu")) return false;
            if (s.Contains("thông tin tài liệu") &&
      s.Contains("trích yếu nội dung") &&
      !s.Contains("quyết định") &&
      !s.Contains("báo cáo") &&
      !s.Contains("kế hoạch") &&
      !s.Contains("công văn") &&
      !s.Contains("thông báo") &&
      !Regex.IsMatch(s, @"\bsố\s*\d{1,6}\s*[-–—]"))
            {
                return false;
            }

            if (s.Length < 30) return false;

            if (s.Contains("đảng cộng sản việt nam") ||
                s.Contains("dang cong san viet nam") ||
                s.Contains("quyết định") ||
                s.Contains("quyet dinh") ||
                s.Contains("báo cáo") ||
                s.Contains("bao cao") ||
                s.Contains("kế hoạch") ||
                s.Contains("ke hoach") ||
                s.Contains("thông báo") ||
                s.Contains("thong bao") ||
                s.Contains("công văn") ||
                s.Contains("cong van") ||
                s.Contains("căn cứ") ||
                s.Contains("can cu") ||
                Regex.IsMatch(s, @"\bsố\s*\d{1,6}\s*[-–—]") ||
                Regex.IsMatch(s, @"\bso\s*\d{1,6}\s*[-–—]"))
            {
                return true;
            }

            return false;
        }


        private string NormalizeOcrText(string s)
        {
            if (s == null) return "";

            s = s.Replace("\r", " ").Replace("\n", " ");
            s = Regex.Replace(s, @"\s+", " ").Trim();

            // Sửa vài lỗi OCR thường gặp
            s = s.Replace("SỐ", "Số");
            s = s.Replace("SO ", "Số ");
            s = s.Replace("S0 ", "Số ");

            return s;
        }

        private IWebElement FindEditableFieldByExactLabel(string label)
        {
            string labelXpath;

            if (label == "Số văn bản")
            {
                // Bắt đúng label "Số văn bản", KHÔNG bắt "Số thứ tự văn bản..."
                labelXpath =
                    "//*[self::label or self::span or self::div or self::td]" +
                    "[normalize-space()='Số văn bản' or normalize-space()='Số văn bản (*)']";
            }
            else if (label == "Ký hiệu văn bản")
            {
                labelXpath =
                    "//*[self::label or self::span or self::div or self::td]" +
                    "[normalize-space()='Ký hiệu văn bản' or normalize-space()='Ký hiệu văn bản (*)']";
            }
            else if (label == "Tác giả văn bản")
            {
                labelXpath =
                    "//*[self::label or self::span or self::div or self::td]" +
                    "[contains(normalize-space(),'Tác giả văn bản')]";
            }
            else
            {
                labelXpath =
                    "//*[self::label or self::span or self::div or self::td]" +
                    "[contains(normalize-space(),'" + label + "')]";
            }

            var labels = driver.FindElements(By.XPath(labelXpath));

            foreach (var lb in labels)
            {
                try
                {
                    if (!lb.Displayed) continue;

                    IWebElement field = lb.FindElement(By.XPath(
                        "./following::input[not(@type='hidden') and not(@readonly) and not(@disabled)][1] | " +
                        "./following::textarea[not(@readonly) and not(@disabled)][1]"
                    ));

                    if (field != null && field.Displayed && field.Enabled)
                        return field;
                }
                catch
                {
                }
            }

            // Fallback theo vị trí form trong trang hiện tại
            if (label == "Số văn bản")
            {
                var fields = driver.FindElements(By.XPath(
                    "//input[not(@type='hidden') and not(@readonly) and not(@disabled)]"
                ));

                foreach (var f in fields)
                {
                    try
                    {
                        string val = f.GetAttribute("value") ?? "";
                        string ph = f.GetAttribute("placeholder") ?? "";

                        if (f.Displayed && f.Enabled &&
                            !ph.Contains("thứ tự") &&
                            !val.StartsWith("A29.125"))
                        {
                            return f;
                        }
                    }
                    catch
                    {
                    }
                }
            }

            throw new Exception("Không tìm thấy ô nhập liệu cho label: " + label);
        }

        private string InferTenTheLoaiVanBan(string kyHieuVanBan, string rawOcrText)
        {
            // Ưu tiên 1: ký hiệu văn bản — chắc chắn nhất
            // Nếu ký hiệu là CV/ĐU thì bắt buộc là Công văn, dù OCR nội dung có "Chương trình", "Kế hoạch"...
            string fromKyHieu = InferTenTheLoaiFromKyHieu(kyHieuVanBan);
            if (!string.IsNullOrWhiteSpace(fromKyHieu))
                return fromKyHieu;

            // Ưu tiên 2: dòng tiêu đề lớn trong OCR (QUYẾT ĐỊNH, BÁO CÁO...) — chỉ dùng khi không có ký hiệu
            string fromText = InferTenTheLoaiFromOcrTitle(rawOcrText);
            if (!string.IsNullOrWhiteSpace(fromText))
                return fromText;

            return "";
        }
        private string InferTenTheLoaiFromOcrTitle(string rawOcrText)
        {
            if (string.IsNullOrWhiteSpace(rawOcrText))
                return "";

            string normalized = rawOcrText
                .Replace("\r", "\n")
                .Replace("\u00A0", " ");

            normalized = Regex.Replace(normalized, @"[ \t]+", " ");

            var lines = normalized
                .Split(new[] { '\n' }, StringSplitOptions.RemoveEmptyEntries)
                .Select(x => CleanTrichYeuLine(x))
                .Where(x => !string.IsNullOrWhiteSpace(x))
                .Where(x => !IsUiNoiseLine(x))
                .ToList();

            foreach (string line in lines)
            {
                string s = Regex.Replace(line.Trim(), @"\s+", " ");
                string upper = s.ToUpperInvariant();

                // Chỉ nhận dòng tiêu đề văn bản thật, không nhận nội dung thân văn bản.
                if (Regex.IsMatch(upper, @"^QUYẾT\s*ĐỊNH\b|^QUYẾT\s*ĐINH\b|^QUYET\s*DINH\b"))
                    return "Quyết định";

                if (Regex.IsMatch(upper, @"^BÁO\s*CÁO\b|^BAO\s*CAO\b"))
                    return "Báo cáo";

                if (Regex.IsMatch(upper, @"^KẾ\s*HOẠCH\b|^KE\s*HOACH\b"))
                    return "Kế hoạch";

                if (Regex.IsMatch(upper, @"^THÔNG\s*BÁO\b|^THONG\s*BAO\b"))
                    return "Thông báo";

                if (Regex.IsMatch(upper, @"^TỜ\s*TRÌNH\b|^TO\s*TRINH\b"))
                    return "Tờ trình";

                if (Regex.IsMatch(upper, @"^NGHỊ\s*QUYẾT\b|^NGHI\s*QUYET\b"))
                    return "Nghị quyết";

                if (Regex.IsMatch(upper, @"^CÔNG\s*VĂN\b|^CONG\s*VAN\b"))
                    return "Công văn";

                if (Regex.IsMatch(upper, @"^CHƯƠNG\s*TRÌNH\b|^CHUONG\s*TRINH\b"))
                    return "Chương trình";

                if (Regex.IsMatch(upper, @"^QUY\s*CHẾ\b|^QUY\s*CHE\b"))
                    return "Quy chế";

                if (Regex.IsMatch(upper, @"^HƯỚNG\s*DẪN\b|^HUONG\s*DAN\b"))
                    return "Hướng dẫn";

                if (Regex.IsMatch(upper, @"^BIÊN\s*BẢN\b|^BIEN\s*BAN\b"))
                    return "Biên bản";

                if (Regex.IsMatch(upper, @"^CHỈ\s*THỊ\b|^CHI\s*THI\b"))
                    return "Chỉ thị";

                if (Regex.IsMatch(upper, @"^ĐỀ\s*ÁN\b|^DE\s*AN\b"))
                    return "Đề án";
            }

            return "";
        }

        private string InferTenTheLoaiFromKyHieu(string kyHieuVanBan)
        {
            if (string.IsNullOrWhiteSpace(kyHieuVanBan))
                return "";

            string s = kyHieuVanBan.Trim().ToUpperInvariant();

            // Nếu truyền vào cả cụm dạng "237-CV/ĐU" thì lấy phần sau dấu gạch.
            Match full = Regex.Match(s,
                @"\d{1,6}\s*[-–—]\s*(?<code>[A-ZĐÂĂÊÔƠƯ0-9]+)",
                RegexOptions.IgnoreCase);

            string code;

            if (full.Success)
            {
                code = full.Groups["code"].Value;
            }
            else
            {
                // Nếu chỉ có "CV/ĐU" thì lấy phần trước dấu /
                code = s.Split('/')[0];
            }

            code = code.Trim();
            code = code.Replace(".", "");
            code = code.Replace(" ", "");

            switch (code)
            {
                case "CV":
                    return "Công văn";

                case "QĐ":
                case "QD":
                    return "Quyết định";

                case "BC":
                    return "Báo cáo";

                case "KH":
                    return "Kế hoạch";

                case "TB":
                    return "Thông báo";

                case "TTR":
                case "TT":
                    return "Tờ trình";

                case "NQ":
                    return "Nghị quyết";

                case "CT":
                    return "Chỉ thị";

                case "HD":
                case "HĐ":
                    return "Hướng dẫn";

                case "BB":
                    return "Biên bản";

                case "QC":

                case "CTR":
                case "CTr":

            }

            return "";
        }

        private bool DocTypeEquals(string a, string b)
        {
            string x = NormalizeDocTypeText(a);
            string y = NormalizeDocTypeText(b);

            if (string.IsNullOrWhiteSpace(x) || string.IsNullOrWhiteSpace(y))
                return false;

            return x == y || x.Contains(y) || y.Contains(x);
        }

        private string NormalizeDocTypeText(string s)
        {
            if (string.IsNullOrWhiteSpace(s))
                return "";

            s = s.Trim().ToLowerInvariant();
            s = Regex.Replace(s, @"\s+", " ");

            s = s.Replace("đ", "d");

            s = s.Replace("á", "a").Replace("à", "a").Replace("ả", "a").Replace("ã", "a").Replace("ạ", "a");
            s = s.Replace("ă", "a").Replace("ắ", "a").Replace("ằ", "a").Replace("ẳ", "a").Replace("ẵ", "a").Replace("ặ", "a");
            s = s.Replace("â", "a").Replace("ấ", "a").Replace("ầ", "a").Replace("ẩ", "a").Replace("ẫ", "a").Replace("ậ", "a");

            s = s.Replace("é", "e").Replace("è", "e").Replace("ẻ", "e").Replace("ẽ", "e").Replace("ẹ", "e");
            s = s.Replace("ê", "e").Replace("ế", "e").Replace("ề", "e").Replace("ể", "e").Replace("ễ", "e").Replace("ệ", "e");

            s = s.Replace("í", "i").Replace("ì", "i").Replace("ỉ", "i").Replace("ĩ", "i").Replace("ị", "i");

            s = s.Replace("ó", "o").Replace("ò", "o").Replace("ỏ", "o").Replace("õ", "o").Replace("ọ", "o");
            s = s.Replace("ô", "o").Replace("ố", "o").Replace("ồ", "o").Replace("ổ", "o").Replace("ỗ", "o").Replace("ộ", "o");
            s = s.Replace("ơ", "o").Replace("ớ", "o").Replace("ờ", "o").Replace("ở", "o").Replace("ỡ", "o").Replace("ợ", "o");

            s = s.Replace("ú", "u").Replace("ù", "u").Replace("ủ", "u").Replace("ũ", "u").Replace("ụ", "u");
            s = s.Replace("ư", "u").Replace("ứ", "u").Replace("ừ", "u").Replace("ử", "u").Replace("ữ", "u").Replace("ự", "u");

            s = s.Replace("ý", "y").Replace("ỳ", "y").Replace("ỷ", "y").Replace("ỹ", "y").Replace("ỵ", "y");

            return s.Trim();
        }

        {
            SelectElement select = new SelectElement(selectElement);

        {
            IWebElement labelElement = driver.FindElement(By.XPath(

            ));

            var candidates = labelElement.FindElements(By.XPath(
                "./following::*[" +
                "contains(@class,'select') or " +
                "contains(@class,'dropdown') or " +
                "contains(@class,'ng-select') or " +
                "contains(@class,'select2') or " +
                "@role='combobox' or " +

                "//*[contains(@class,'option') or contains(@class,'item') or @role='option' or self::li or self::span or self::div]" +
                "[contains(normalize-space(),'" + text + "')]"

            option.Click();

        private IWebElement WaitVisible(By by)
        {
            return wait.Until(d =>
            {
                var elements = d.FindElements(by);

                foreach (var element in elements)
                {
                    try
                    {
                        if (element.Displayed)
                            return element;
                    }
                    catch
                    {
                    }
                }

                return null;
            });
        }

        private IWebElement WaitClickable(By by)
        {
            return wait.Until(d =>
            {
                var elements = d.FindElements(by);

                foreach (var element in elements)
                {
                    try
                    {
                        if (element.Displayed && element.Enabled)
                            return element;
                    }
                    catch
                    {
                    }
                }

                return null;
            });
        }

        private void AddLog(string step, string status, string detail)
        {
            if (gridLog == null) return;

            int rowIndex = gridLog.Rows.Add(
                DateTime.Now.ToString("HH:mm:ss"),
                step,
                status,
                detail
            );

            DataGridViewRow row = gridLog.Rows[rowIndex];

            if (status == "OK")
                row.DefaultCellStyle.BackColor = Color.FromArgb(220, 255, 220);
            else if (status == "FAILED")
                row.DefaultCellStyle.BackColor = Color.FromArgb(255, 220, 220);
            else if (status == "Cảnh báo")
                row.DefaultCellStyle.BackColor = Color.FromArgb(255, 245, 200);
            else if (status == "Đang làm")
                row.DefaultCellStyle.BackColor = Color.FromArgb(220, 235, 255);

            if (!FAST_MODE || status == "FAILED" || status == "Cảnh báo" || logCounter % 5 == 0)
            {
                gridLog.FirstDisplayedScrollingRowIndex = rowIndex;
                gridLog.Refresh();
                Application.DoEvents();
            }
        }

        private bool TryExtractSoKyHieuVanBan(string rawText, out string soVanBan, out string kyHieuVanBan)
        {
            soVanBan = "";
            kyHieuVanBan = "";

            if (string.IsNullOrWhiteSpace(rawText))
                return false;

            string text = rawText
                .Replace("\r", "\n")
                .Replace("\u00A0", " ");

            text = Regex.Replace(text, @"[ \t]+", " ");

            var lines = text
                .Split(new[] { '\n' }, StringSplitOptions.RemoveEmptyEntries)
                .Select(x => Regex.Replace(x.Trim(), @"\s+", " "))
                .Where(x => !string.IsNullOrWhiteSpace(x))
                .ToList();

            if (lines.Count == 0)
                return false;

            string expectedDocType = InferTenTheLoaiFromOcrTitle(rawText);

            int docTitleIndex = -1;
            int scanLimitForTitle = Math.Min(lines.Count, 80);

            for (int i = 0; i < scanLimitForTitle; i++)
            {
                if (IsDocumentTitleLine(lines[i]))
                {
                    docTitleIndex = i;
                    break;
                }
            }

            int bestScore = int.MinValue;
            string bestSo = "";
            string bestKy = "";
            string bestLine = "";

            int scanLimit = Math.Min(lines.Count, 80);

            for (int i = 0; i < scanLimit; i++)
            {
                string line = lines[i];

                // Bắt các dạng:
                // Số 32-KH/ĐU
                // Số: 32 - KH/ĐU
                // 32-KH/ĐU
                // Nhưng sau đó sẽ chấm điểm để loại "Chỉ thị số 28-CT/TW" trong trích yếu.
                MatchCollection matches = Regex.Matches(
                    line,
                    @"(?:(?:^|[\s:：])S[oố]\s*[:：]?\s*)?(?<so>\d{1,6})\s*[-–—]\s*(?<ky>[A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]{1,10}(?:\s*[\/\-]\s*[A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]{1,15})+)",
                    RegexOptions.IgnoreCase
                );

                foreach (Match m in matches)
                {
                    if (!m.Success)
                        continue;

                    string so = m.Groups["so"].Value.Trim();
                    string ky = NormalizeKyHieu(m.Groups["ky"].Value);
                    string candidateType = InferTenTheLoaiFromKyHieu(ky);

                    if (string.IsNullOrWhiteSpace(so) || string.IsNullOrWhiteSpace(ky))
                        continue;

                    string keyLine = FoldVietnameseKey(line);
                    int score = 0;

                    // Dòng header thường bắt đầu bằng "Số".
                    if (Regex.IsMatch(keyLine, @"^SO\s*\d{1,6}\s*[-–—]", RegexOptions.IgnoreCase) ||
                        Regex.IsMatch(keyLine, @"^SO\s*[:：]?\s*\d{1,6}\s*[-–—]", RegexOptions.IgnoreCase))
                    {
                        score += 50;
                    }
                    else
                    {
                        score -= 20;
                    }

                    // Số/Ký hiệu header thường nằm trước hoặc rất gần dòng loại văn bản lớn.
                    if (docTitleIndex >= 0)
                    {
                        if (i <= docTitleIndex + 1)
                            score += 45;
                        else
                            score -= 25;
                    }

                    // Quan trọng nhất:
                    // Nếu PDF có dòng KẾ HOẠCH thì phải chọn ký hiệu loại KH,
                    // không được chọn CT/TW trong cụm "Chỉ thị số 28-CT/TW".
                    if (!string.IsNullOrWhiteSpace(expectedDocType) &&
                        !string.IsNullOrWhiteSpace(candidateType))
                    {
                        if (DocTypeEquals(candidateType, expectedDocType))
                            score += 120;
                        else
                            score -= 150;
                    }

                    // Văn bản của Đảng ủy phường thường có /ĐU.
                    if (ky.ToUpperInvariant().Contains("/ĐU") || ky.ToUpperInvariant().Contains("/DU"))
                        score += 30;

                    // Loại các số nằm trong trích yếu/thân văn bản.
                    if (keyLine.Contains("CHI THI SO") ||
                        keyLine.Contains("KE HOACH SO") ||
                        keyLine.Contains("CONG VAN SO") ||
                        keyLine.Contains("QUYET DINH SO"))
                    {
                        score -= 90;
                    }

                    // Dòng có ngày/tháng/năm thường là trích yếu/thân, không phải dòng Số header.
                    if (keyLine.Contains(" NGAY ") &&
                        keyLine.Contains(" THANG ") &&
                        keyLine.Contains(" NAM "))
                    {
                        score -= 35;
                    }

                    if (score > bestScore)
                    {
                        bestScore = score;
                        bestSo = so;
                        bestKy = ky;
                        bestLine = line;
                    }
                }
            }

            // Nếu điểm quá thấp thì không nhận, tránh lấy nhầm số trong nội dung.
            if (bestScore < 0)
            {
                AddLog("Đọc số/ký hiệu", "Cảnh báo",
                    "Có thấy mẫu số/ký hiệu nhưng điểm header thấp, không dùng. Candidate: " + bestLine);
                return false;
            }

            soVanBan = bestSo;
            kyHieuVanBan = bestKy;

            AddLog("Đọc số/ký hiệu", "OK",
                "Chọn header theo điểm: Số = " + soVanBan +
                ", Ký hiệu = " + kyHieuVanBan +
                ", Điểm = " + bestScore +
                ", Dòng = " + bestLine);

            return true;
        }
        // Hàm chuẩn hóa ký hiệu văn bản tách riêng để tái sử dụng
        private string NormalizeKyHieu(string raw)
        {
            if (string.IsNullOrWhiteSpace(raw)) return "";

            string s = raw.Trim()
                .Replace(" ", "")
                .Replace("\\", "/")
                .Trim('-', '–', '—', '.', ',', ':', ';');

            // "CV-DU" → "CV/ĐU"
            s = Regex.Replace(s,
                @"^([A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]{1,10})-([A-ZĐÂĂÊÔƠƯÀ-Ỹ0-9]{1,15})$",
                "$1/$2");

            // "/DU" → "/ĐU"
            s = Regex.Replace(s, @"/DU\b", "/ĐU", RegexOptions.IgnoreCase);
            s = Regex.Replace(s, @"-DU\b", "/ĐU", RegexOptions.IgnoreCase);

            return s;
        }
        private string TryFixHorizontalBadOcrTitle(string rawOcrText)
        {
            if (string.IsNullOrWhiteSpace(rawOcrText))
                return "";

            string norm = NormalizeVietnameseLoose(rawOcrText);

            bool isBadHorizontalNoiChinh =
                  norm.Contains("bao cao ngay") &&
                  norm.Contains("cong tac") &&
                  (
                      norm.Contains("phung chong") ||   // OCR sai "Phùng chống"
                      norm.Contains("phung thong")       // OCR sai "Phùng thông"
                                                         // KHÔNG bắt "phong chong" vì đó là chính tả đúng
                  ) &&
                  (
                    norm.Contains("lang phi ti") ||
                    norm.Contains("lang phi tieu") ||
                    norm.Contains("l ang phi")
                ) &&
                (
                    norm.Contains("cai cach tu phap") ||
                    norm.Contains("cai cachtn phap") ||
                    norm.Contains("tu phap nhu sau")
                );

            if (isBadHorizontalNoiChinh)
            {
                return "VỀ CÔNG TÁC NỘI CHÍNH, PHÒNG CHỐNG THAM NHŨNG, LÃNG PHÍ TIÊU CỰC VÀ CẢI CÁCH TƯ PHÁP";
            }

            return "";
        }

        private string CleanTitleAfterLocalOcr(string value)
        {
            if (string.IsNullOrWhiteSpace(value))
                return "";

            string s = value.Trim();

            // Cắt phần thân bị OCR nối vào sau trích yếu.
            // Ví dụ sai:
            // "... đưa những đảng viên không còn đủ tư cách ra khỏi Đảng Thực hiện Chỉ thị số 28-CT/TW; Kế hoạch số..."
            //
            // Không cắt cụm đúng:
            // "tiếp tục thực hiện Chỉ thị số 28-CT/TW ngày..."
            // vì cụm đúng không có dấu ; ngay sau CT/TW.
            s = Regex.Replace(
                s,
                @"\s+Thực\s+hiện\s+Chỉ\s+thị\s+số\s+\d{1,6}\s*[-–—]\s*CT\s*/\s*TW\s*;.*$",
                "",
                RegexOptions.IgnoreCase
            );

            s = Regex.Replace(
                s,
                @"\s+Thuc\s+hien\s+Chi\s+thi\s+so\s+\d{1,6}\s*[-–—]\s*CT\s*/\s*TW\s*;.*$",
                "",
                RegexOptions.IgnoreCase
            );

            s = Regex.Replace(s, @"\s+", " ").Trim();
            s = CleanFinalTrichYeuTitle(s);

            return s;
        }

        private bool LooksLikeTextCanExtractTrichYeu(string text)
        {
            if (string.IsNullOrWhiteSpace(text)) return false;

            string s = text.ToLowerInvariant();
            s = Regex.Replace(s, @"\s+", " ").Trim();

            if (s.Length < 20) return false;

            // Cho phép OCR thiếu số/ký hiệu, miễn là có dòng trích yếu dạng "Về..."
            if (Regex.IsMatch(text,
                @"(?im)^\s*(Về|V/v|V\/v|Về việc)\b"))
                return true;

            return false;
        }
